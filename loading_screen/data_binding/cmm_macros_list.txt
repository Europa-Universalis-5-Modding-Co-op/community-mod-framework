# Args: SettingKey (string), SlotKey (string)
# Returns: CFixedPoint
macro = {
    description = "Normalized slider maximum for one list field slot (max minus min)"
    definition = "CMMListFieldSliderNormalizedMax(SettingKey,SlotKey)"
    replace_with = "Subtract_CFixedPoint(CMMListFieldMetadataValue(SettingKey, SlotKey, '_max_'), CMMListFieldMetadataValue(SettingKey, SlotKey, '_min_'))"
}

# Args: ItemKey (string), SlotKey (string)
# Returns: CFixedPoint
macro = {
    description = "Normalized slider value for one list item field (value minus min)"
    definition = "CMMListFieldSliderNormalizedValue(ItemKey,SlotKey)"
    replace_with = "Subtract_CFixedPoint(CMMListItemFieldValue(ItemKey, SlotKey), CMMListFieldMetadataValue(CMMListItemOwnerSettingKey(ItemKey), SlotKey, '_min_'))"
}

# Args: ItemKey (string), SlotKey (string)
# Returns: int32
macro = {
    description = "Slider thumb visual index on 0-100 scale for one list item field"
    definition = "CMMListSliderThumbIndex(ItemKey,SlotKey)"
    replace_with = "FixedPointToInt(Add_CFixedPoint(Multiply_CFixedPoint(Divide_CFixedPoint(CMMListFieldSliderNormalizedValue(ItemKey, SlotKey), CMMListFieldSliderNormalizedMax(CMMListItemOwnerSettingKey(ItemKey), SlotKey)), '(CFixedPoint)100'), '(CFixedPoint)0.500'))"
}
