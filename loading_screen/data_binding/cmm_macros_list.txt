# Args: SettingKey (string), ItemKey (string), SlotKey (string)
# Returns: string
macro = {
    description = "Sign-conditional prefix: selects _prefix_high (>0), _prefix_low (<0), or _prefix (==0) without existence checks"
    definition = "CMMListFieldConditionalPrefix(SettingKey,ItemKey,SlotKey)"
    replace_with = "Select_CString(GreaterThan_CFixedPoint(CMMListItemFieldValue(ItemKey, SlotKey), '(CFixedPoint)0'), CMMListFieldPrefixHigh(SettingKey, SlotKey), Select_CString(LessThan_CFixedPoint(CMMListItemFieldValue(ItemKey, SlotKey), '(CFixedPoint)0'), CMMListFieldPrefixLow(SettingKey, SlotKey), AddTextIf(CMMListFieldHasPrefix(SettingKey, SlotKey), CMMListFieldPrefix(SettingKey, SlotKey))))"
}

# Args: SettingKey (string), ItemKey (string), SlotKey (string)
# Returns: string
macro = {
    description = "Sign-conditional postfix: selects _postfix_high (>0), _postfix_low (<0), or _postfix (==0) without existence checks"
    definition = "CMMListFieldConditionalPostfix(SettingKey,ItemKey,SlotKey)"
    replace_with = "Select_CString(GreaterThan_CFixedPoint(CMMListItemFieldValue(ItemKey, SlotKey), '(CFixedPoint)0'), CMMListFieldPostfixHigh(SettingKey, SlotKey), Select_CString(LessThan_CFixedPoint(CMMListItemFieldValue(ItemKey, SlotKey), '(CFixedPoint)0'), CMMListFieldPostfixLow(SettingKey, SlotKey), AddTextIf(CMMListFieldHasPostfix(SettingKey, SlotKey), CMMListFieldPostfix(SettingKey, SlotKey))))"
}

# Args: ItemKey (string), SlotKey (string)

    description = "Slider thumb visual index on 0-100 scale for one list item field"
    definition = "CMMListSliderThumbIndex(ItemKey,SlotKey)"
    replace_with = "FixedPointToInt(Add_CFixedPoint(Multiply_CFixedPoint(Divide_CFixedPoint(CMMListFieldSliderNormalizedValue(ItemKey, SlotKey), CMMListFieldSliderNormalizedMax(CMMListItemOwnerSettingKey(ItemKey), SlotKey)), '(CFixedPoint)100'), '(CFixedPoint)0.500'))"
}
