# CMM GUI Macros: View State And Selection

# Returns: boolean
macro = {
    description = "True when selected mod variable-system key exists"
    definition = "CMMSelectedModExists"
    replace_with = "CMMSelectedValueExists('cmm_selected_mod')"
}

# Returns: boolean
macro = {
    description = "True when selected tab variable-system key exists"
    definition = "CMMSelectedTabExists"
    replace_with = "CMMSelectedValueExists('cmm_selected_tab')"
}

# Returns: boolean
macro = {
    description = "True when no mods are registered yet"
    definition = "CMMNoModsRegistered"
    replace_with = "IsDataModelEmpty(CMMHomeScope.GetList('cmm_registered_mod_ids'))"
}

# Returns: boolean
macro = {
    description = "True when at least one mod is registered"
    definition = "CMMHasRegisteredMods"
    replace_with = "Not(IsDataModelEmpty(CMMHomeScope.GetList('cmm_registered_mod_ids')))"
}

# Returns: boolean
macro = {
    description = "Show empty-state text when selected tab has zero visible settings"
    definition = "CMMShowNoSettingsForSelectedTab"
    replace_with = "And(CMMSelectedModExists, Or(Not(CMMSelectedTabExists), Not(CMMTabNonEmpty(GetVariableSystem.Get('cmm_selected_tab')))))"
}

# Tab hierarchy, declared with cmm_register_subtab.

# Args: TabKey (string)
# Returns: boolean
macro = {
    description = "True when one tab is a sub-tab of another tab"
    definition = "CMMTabHasParent(TabKey)"
    replace_with = "CMMHomeScope.GetVariableFromVariableMap('cmm_tab_parent', MakeScopeFlag(TabKey)).IsSet"
}

# Args: TabKey (string)
# Returns: string
macro = {
    description = "Parent tab key for one sub-tab"
    definition = "CMMTabParent(TabKey)"
    replace_with = "CMMHomeScope.GetVariableFromVariableMap('cmm_tab_parent', MakeScopeFlag(TabKey)).GetFlagName"
}

# Args: TabKey (string)
# Returns: boolean
macro = {
    description = "True when one tab has sub-tabs"
    definition = "CMMTabHasChildren(TabKey)"
    replace_with = "CMMHomeScope.GetVariableFromVariableMap('cmm_tab_default_child', MakeScopeFlag(TabKey)).IsSet"
}

# Args: TabKey (string)
# Returns: string
macro = {
    description = "First sub-tab key for one tab"
    definition = "CMMTabDefaultChild(TabKey)"
    replace_with = "CMMHomeScope.GetVariableFromVariableMap('cmm_tab_default_child', MakeScopeFlag(TabKey)).GetFlagName"
}

# Args: TabKey (string)
# Returns: string
macro = {
    description = "Last sub-tab opened under one tab"
    definition = "CMMTabLastChild(TabKey)"
    replace_with = "GetVariableSystem.Get(Concatenate(TabKey, '_lsub'))"
}

# Args: TabKey (string)
# Returns: boolean
macro = {
    description = "True when one tab's remembered sub-tab is still one of its sub-tabs and still has settings"
    definition = "CMMTabLastChildValid(TabKey)"
    replace_with = "And3(CMMSelectedValueExists(Concatenate(TabKey, '_lsub')), EqualTo_string(CMMTabParent(CMMTabLastChild(TabKey)), TabKey), CMMTabNonEmpty(CMMTabLastChild(TabKey)))"
}

# Args: TabKey (string)
# Returns: string
macro = {
    description = "Sub-tab one tab opens on: the last one opened under it, otherwise its first"
    definition = "CMMTabOpenChild(TabKey)"
    replace_with = "Select_CString(CMMTabLastChildValid(TabKey), CMMTabLastChild(TabKey), CMMTabDefaultChild(TabKey))"
}

# Args: TabKey (string)
# Returns: string
macro = {
    description = "Settings-bearing tab key for one tab, resolving a tab with sub-tabs to the sub-tab it opens on"
    definition = "CMMTabLeaf(TabKey)"
    replace_with = "Select_CString(CMMTabHasChildren(TabKey), CMMTabOpenChild(TabKey), TabKey)"
}

# Returns: string
macro = {
    description = "Top-row tab key for the current selection"
    definition = "CMMSelectedTopTab"
    replace_with = "Select_CString(CMMTabHasParent(GetVariableSystem.Get('cmm_selected_tab')), CMMTabParent(GetVariableSystem.Get('cmm_selected_tab')), GetVariableSystem.Get('cmm_selected_tab'))"
}

# Returns: boolean
macro = {
    description = "True when the selected mod's selected top-row tab has sub-tabs to show"
    definition = "CMMSubTabRowShown"
    replace_with = "And(CMMSelectedModExists, CMMTabHasChildren(CMMSelectedTopTab))"
}

# Collapsed-group state, toggled from the group header.

# Args: GroupKey (string)
# Returns: boolean
macro = {
    description = "True when one group is collapsed"
    definition = "CMMGroupCollapsed(GroupKey)"
    replace_with = "CMMValueEqualsOne(CMMMapValue('cmm_group_collapsed', GroupKey))"
}

# Args: GroupKey (string)
# Returns: boolean
macro = {
    description = "True when one group is expanded"
    definition = "CMMGroupExpanded(GroupKey)"
    replace_with = "Not(CMMGroupCollapsed(GroupKey))"
}

# Args: GroupKey (string)
# Returns: none
macro = {
    description = "Toggle whether one group renders its settings box"
    definition = "CMMToggleGroupCollapsed(GroupKey)"
    replace_with = "GetScriptedGui('CMM_ToggleGroupCollapsed').Execute(GuiScope.SetRoot(CMMHomeScope).AddScope('cmm_group', MakeScopeFlag(GroupKey)).End)"
}

# Args: GroupKey (string)
# Returns: none
macro = {
    description = "Apply the state one group is about to take to every group in its tab"
    definition = "CMMToggleTabGroupsCollapsed(GroupKey)"
    replace_with = "GetScriptedGui('CMM_ToggleTabGroupsCollapsed').Execute(GuiScope.SetRoot(CMMHomeScope).AddScope('cmm_group', MakeScopeFlag(GroupKey)).End)"
}

# Args: SettingKey (string)
# Returns: variable
macro = {
    description = "Mirrored is_shown recorded on the home country for one conditional setting"
    definition = "CMMRollupRecordedShown(SettingKey)"
    replace_with = "CMMHomeScope.GetVariableFromVariableMap('cmm_setting_shown', MakeScopeFlag(SettingKey))"
}

# Compared against the country's own record rather than a GUI-side copy of it, so a probe
# re-fires whenever that record is missing or disagrees: on a country change, on a setting
# never probed on this country, and on an ordinary is_shown flip.
# Args: SettingKey (string)
# Returns: boolean
macro = {
    description = "True when the home country's record matches one conditional setting's current is_shown"
    definition = "CMMRollupRecordIsCurrent(SettingKey)"
    replace_with = "And(CMMRollupRecordedShown(SettingKey).IsSet, EqualTo_CFixedPoint(CMMRollupRecordedShown(SettingKey).GetValue, Select_CFixedPoint(CMMGuiIsShown(Concatenate(SettingKey, '_on_changed')), '(CFixedPoint)1', '(CFixedPoint)0')))"
}

# Args: SettingKey (string)
# Returns: none
macro = {
    description = "Mirror one conditional setting's is_shown and rebuild the rollup"
    definition = "CMMRollupMarkSetting(SettingKey)"
    replace_with = "GetScriptedGui(Select_CString(CMMGuiIsShown(Concatenate(SettingKey, '_on_changed')), 'CMM_MarkSettingShown', 'CMM_MarkSettingHidden')).Execute(GuiScope.SetRoot(CMMHomeScope).AddScope('setting', MakeScopeFlag(SettingKey)).End)"
}
