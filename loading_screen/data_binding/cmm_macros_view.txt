# CMM GUI Macros: View State And Selection

# Returns: boolean
macro = {
    description = "True when selected mod variable-system key exists"
    definition = "CMMSelectedModExists"
    replace_with = "CMMSelectedValueExists('cmm_selected_mod')"
}

# Returns: boolean
macro = {
    description = "True when selected tab variable-system key exists"
    definition = "CMMSelectedTabExists"
    replace_with = "CMMSelectedValueExists('cmm_selected_tab')"
}

# Args: GroupKey (string)
# Returns: boolean
macro = {
    description = "Visibility rule for one group row"
    definition = "CMMGroupRowVisible(GroupKey)"
    replace_with = "CMMMatchesSelectedModAndTab(CMMMetadataFlag('cmm_group_owner_mod_id_', GroupKey), CMMMetadataFlag('cmm_group_owner_tab_key_', GroupKey))"
}

# Returns: boolean
macro = {
    description = "True when no mods are registered yet"
    definition = "CMMNoModsRegistered"
    replace_with = "IsDataModelEmpty(Player.MakeScope.GetList('cmm_registered_mod_ids'))"
}

# Returns: boolean
macro = {
    description = "True when at least one mod is registered"
    definition = "CMMHasRegisteredMods"
    replace_with = "Not(IsDataModelEmpty(Player.MakeScope.GetList('cmm_registered_mod_ids')))"
}

# Args: ModId (string)
# Returns: boolean
macro = {
    description = "Filter match for one mod row against current search text"
    definition = "CMMModMatchesSearch(ModId)"
    replace_with = "Or4(Not(CMMSelectedValueExists('cmm_mod_search_text')), StringIsEmpty(GetVariableSystem.Get('cmm_mod_search_text')), StringContains(CMMLocalizedSuffix(ModId, '_name'), GetVariableSystem.Get('cmm_mod_search_text')), StringContains(ModId, GetVariableSystem.Get('cmm_mod_search_text')))"
}

# Returns: boolean
macro = {
    description = "Show empty-state text when selected tab has zero visible settings"
    definition = "CMMShowNoSettingsForSelectedTab"
    replace_with = "And(CMMSelectedModExists, Or(Not(CMMSelectedTabExists), Not(CMMTabNonEmpty(GetVariableSystem.Get('cmm_selected_tab')))))"
}

# Cached group-row metadata, written once per group by the cmm_cache_group_row state.

# Args: GroupKey (string)
# Returns: boolean
macro = {
    description = "Cached visibility rule for one group row"
    definition = "CMMCachedGroupRowVisible(GroupKey)"
    replace_with = "CMMMatchesSelectedModAndTab(GetVariableSystem.Get(Concatenate(GroupKey, '_grom')), GetVariableSystem.Get(Concatenate(GroupKey, '_grot')))"
}

macro = {

    definition = "CMMCachedGroupName(GroupKey)"

}

}

# Empty-container rollup, maintained by cmm_rollup_driver and _cmm_recompute_rollup.

# Args: GroupKey (string)
# Returns: boolean
macro = {
    description = "True when one group has at least one currently-visible setting"
    definition = "CMMGroupNonEmpty(GroupKey)"
    replace_with = "Player.MakeScope.GetVariableFromVariableMap('cmm_group_nonempty', MakeScopeFlag(GroupKey)).IsSet"
}

# Args: TabKey (string)
# Returns: boolean
macro = {
    description = "True when one tab has at least one currently-visible setting"
    definition = "CMMTabNonEmpty(TabKey)"
    replace_with = "Player.MakeScope.GetVariableFromVariableMap('cmm_tab_nonempty', MakeScopeFlag(TabKey)).IsSet"
}

# Args: ModId (string)
# Returns: boolean
macro = {
    description = "True when one mod has at least one currently-visible setting"
    definition = "CMMModNonEmpty(ModId)"
    replace_with = "Player.MakeScope.GetVariableFromVariableMap('cmm_mod_nonempty', MakeScopeFlag(ModId)).IsSet"
}

# Args: SettingKey (string)
# Returns: string
macro = {
    description = "'yes'/'no' for one conditional setting's current is_shown"
    definition = "CMMRollupShownString(SettingKey)"
    replace_with = "Select_CString(CMMGuiIsShown(Concatenate(SettingKey, '_on_changed')), 'yes', 'no')"
}

# Args: SettingKey (string)
# Returns: none
macro = {
    description = "Mirror one conditional setting's is_shown and rebuild the rollup"
    definition = "CMMRollupMarkSetting(SettingKey)"
    replace_with = "GetScriptedGui(Select_CString(CMMGuiIsShown(Concatenate(SettingKey, '_on_changed')), 'CMM_MarkSettingShown', 'CMM_MarkSettingHidden')).Execute(GuiScope.SetRoot(GetPlayer.MakeScope).AddScope('setting', MakeScopeFlag(SettingKey)).End)"
}
