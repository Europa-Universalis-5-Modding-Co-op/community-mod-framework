    definition = "CMMOrdinalToWidgetIndex(Index)"

}

# Args: Index (int32)

    definition = "CMFLogIsScopeArgs(Index)"

# Args: Index (int32)
# Returns: boolean
macro = {
    description = "True when arg1 is present on a log entry"
    definition = "CMFLogHasArg1(Index)"
    replace_with = "Not(EqualTo_string(GetVariableFromGlobalVariableMap('cmf_log_arg1', CMFLogKey(Index)).GetFlagName, '_cmf_log_no_arg'))"
}

# Args: Index (int32)
# Returns: boolean
macro = {
    description = "True when arg2 is present on a log entry"
    definition = "CMFLogHasArg2(Index)"
    replace_with = "Not(EqualTo_string(GetVariableFromGlobalVariableMap('cmf_log_arg2', CMFLogKey(Index)).GetFlagName, '_cmf_log_no_arg'))"
}

# Args: Index (int32)
# Returns: string
macro = {
    description = "Resolved arg1 text for a log entry (flag localization or country name)"
    definition = "CMFLogArg1Name(Index)"
    replace_with = "Select_CString(CMFLogIsScopeArgs(Index), GetVariableFromGlobalVariableMap('cmf_log_arg1', CMFLogKey(Index)).GetCountry.GetName, Localize(GetVariableFromGlobalVariableMap('cmf_log_arg1', CMFLogKey(Index)).GetFlagName))"
}

# Args: Index (int32)
# Returns: string
macro = {
    description = "Resolved arg2 text for a log entry (flag localization or country name)"
    definition = "CMFLogArg2Name(Index)"
    replace_with = "Select_CString(CMFLogIsScopeArgs(Index), GetVariableFromGlobalVariableMap('cmf_log_arg2', CMFLogKey(Index)).GetCountry.GetName, Localize(GetVariableFromGlobalVariableMap('cmf_log_arg2', CMFLogKey(Index)).GetFlagName))"
}

# Args: Index (int32)
# Returns: string
macro = {
    description = "Full formatted log entry text: [arg1?] action [arg2?]"
    definition = "CMFLogEntryText(Index)"
    replace_with = "Concatenate(AddTextIf(CMFLogHasArg1(Index), Concatenate(CMFLogArg1Name(Index), ' ')), Concatenate(CMFLogAction(Index), AddTextIf(CMFLogHasArg2(Index), Concatenate(' ', CMFLogArg2Name(Index)))))"
}

# Returns: boolean
macro = {

    replace_with = "GreaterThan_CFixedPoint(GetGlobalVariable('cmf_log_count').GetValue, '(CFixedPoint)0')"
}
