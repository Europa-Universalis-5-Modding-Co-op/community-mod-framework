on_game_start = {
	on_actions = {
		delay = { days = 1 }
		cmf_on_yearly
	}
}

yearly_country_pulse = {
	on_actions = {
		cmf_on_yearly
	}
}

# Yearly event dispatcher
# Scope: global
cmf_on_yearly = {
	effect = {
		every_country = {
			limit = { is_ai = no }
			cmm_register_menu_settings = yes
			trigger_event_silently = {
				on_action = cmf_yearly_human_country_pulse
			}
		}
	}
}

# Unified human country pulse
# Slightly more efficient then each mod checking for players individually
# Scope: country
cmf_yearly_human_country_pulse = {
	on_actions = {
		cmf_on_mod_registration
	}
}

# Integration extension point.
# Scope: country
cmf_on_mod_registration = {
	on_actions = { }
}

# Unified callback extension point. Fires after auto-apply updates a setting,
# an alert is clicked, or an action bar button is clicked.
# var:cmf_callback contains the flag key of the target (setting, alert, or element).
# Scope: country
cmf_on_callback = {
	on_actions = { }
}
