on_game_start = {
	on_actions = {
		_cmf_on_init
		cmf_on_banner_registration
	}
}

# Set global presence flag for cross-mod detection.
# Scope: global
_cmf_on_init = {
	effect = {
		set_global_variable = { name = cmf_active value = yes }
		set_global_variable = { name = _cmf_pending_new_game value = yes }
		if = {
			limit = { NOT = { has_global_variable = cmf_log_count } }
			set_global_variable = { name = cmf_log_count value = 0 }
		}
	}
}

# Lobby banner opt-in extension point. Fires at on_game_start before the
# lobby renders its banners. A mod opts in by appending a leaf
# on_action that calls cmf_register_lobby_banner.
# Scope: none
cmf_on_banner_registration = {
	on_actions = {
		_cmf_on_banner_registration
	}
}

# Register CMF core's own lobby banner.
# Scope: none
_cmf_on_banner_registration = {
	effect = {
		clear_global_variable_list = cmf_lobby_banner_mod_ids
		cmf_register_lobby_banner = { mod_id = cmf_core }
	}
}

# CMM menu settings registration. Runs before cmf_on_mod_registration
# to ensure CMF's own settings appear first in the mod list.
# Scope: country
_cmf_on_cmm_registration = {
	effect = {
		clear_variable_list = cmm_registered_mod_ids
		clear_variable_list = cmm_registered_tab_keys
		clear_variable_list = cmm_registered_group_keys
		clear_variable_list = cmm_registered_setting_keys
		clear_variable_list = cmm_conditional_setting_keys
		_cmm_clear_registration_pass_state = yes
		clear_global_variable_list = cmf_active_mod_ids
		_cmm_register_menu_settings = yes
	}
}

# CMM menu settings registration that depends on other mods having completed
# their own registration first (e.g. action bar element lists). Runs after
# cmf_on_mod_registration.

_cmf_on_cmm_registration_late = {
	effect = {

	}
}

# Post-lobby game start hook. Fires once on a new game when the in-game UI
# is created, after the player has selected their country. Vanilla on_game_start
# fires before the country selection screen, so player-scoped effects
# (is_ai = no) don't work there since no country has been selected yet.
# Scope: none
on_game_start_after_lobby = {
	effect = {
		every_country = {
			limit = { is_ai = no }
			trigger_event_silently = {
				on_action = on_game_start_after_lobby_human_country
			}
		}
		_cmf_capture_game_start_mods = yes
	}
	on_actions = { }
}

# Human country variant of on_game_start_after_lobby.
# Scope: country
on_game_start_after_lobby_human_country = {
	on_actions = {
		_cmf_on_cmm_registration
		cmf_on_mod_registration
		_cmf_on_action_bar_prune
		_cmf_on_cmm_registration_late
	}
}

# Save load hook. Fires every time a save game is loaded (including from
# the country selection screen). Does NOT fire on new games.
# Scope: none
on_game_load = {
	effect = {
		set_global_variable = { name = cmf_active value = yes }
		# Close the saved dispatch gate so stale element names wait for this
		# load's registration pass.
		remove_from_global_variable_map = { name = cmf_action_bar_synced key = flag:cmf_action_bar_synced }
		every_country = {
			limit = { is_ai = no }
			trigger_event_silently = {
				on_action = on_game_load_human_country
			}
		}
	}
	on_actions = {
		cmf_on_banner_registration
	}
}

# Human country variant of on_game_load.
# Scope: country
on_game_load_human_country = {
	on_actions = {
		_cmf_on_cmm_registration
		_cmf_on_cmm_registration_late
	}
}

# Post-lobby save load hook. Fires every time a save game is loaded,
