# Show an alert in the alert bar. Player-hidden and muted alerts stay off the bar.
# Scope: country
# Args: alert (key) - the alert root localization key
cmf_show_alert = {
    hidden_effect = {
        add_to_variable_list = {
            name = cmf_active_alerts
            target = flag:$alert$
        }
    }

    # Benign warning suppression
    cmf_suppress = { v = $alert$ }
}

# Remove an alert from the alert bar.
# Scope: country
# Args: alert (key) - the alert root localization key
cmf_remove_alert = {
    hidden_effect = {
        remove_list_variable = {
            name = cmf_active_alerts
            target = flag:$alert$
        }
        _cmf_alert_clear_hidden = { target = flag:$alert$ }
    }
}

# Remove the alert from the bar when it is left-clicked.
# Scope: country

# Mark an alert player-hidden (map for GUI gates, list for panel datamodels).
# Scope: country
# Args: target (target ref) - the alert flag
_cmf_alert_set_hidden = {
    add_to_variable_map = { name = cmf_alert_hidden key = $target$ value = 1 }
    add_to_variable_list = { name = cmf_hidden_alerts target = $target$ }
}

# Clear an alert's player-hidden state.
# Scope: country
# Args: target (target ref) - the alert flag
_cmf_alert_clear_hidden = {
    remove_from_variable_map = { name = cmf_alert_hidden key = $target$ }
    remove_list_variable = { name = cmf_hidden_alerts target = $target$ }
}

# Stage every alert key for the staleness pass driven by cmf_alert_prune_driver.
# Scope: country
_cmf_alert_queue_prune = {
    clear_variable_list = cmf_alert_prune_queue
    every_in_list = {
        variable = cmf_active_alerts
        save_scope_as = _cmf_ap_alert
        root = { add_to_variable_list = { name = cmf_alert_prune_queue target = scope:_cmf_ap_alert } }
    }
    every_in_list = {
        variable = cmf_hidden_alerts
        save_scope_as = _cmf_ap_alert
        root = { add_to_variable_list = { name = cmf_alert_prune_queue target = scope:_cmf_ap_alert } }
    }
    every_in_list = {
        variable = cmf_muted_alerts
        save_scope_as = _cmf_ap_alert
        root = { add_to_variable_list = { name = cmf_alert_prune_queue target = scope:_cmf_ap_alert } }
    }

    # Benign warning suppression
    cmf_suppress = { v = cmf_alert_prune_queue }
}

# Remove every trace of an alert whose mod is no longer loaded.
# Scope: country
# Args: target (target ref) - the alert flag
_cmf_alert_prune = {
    remove_list_variable = { name = cmf_active_alerts target = $target$ }
    _cmf_alert_clear_hidden = { target = $target$ }
    _cmf_alert_clear_muted = { target = $target$ }
    remove_from_variable_map = { name = cmf_alert_remove_on_click key = $target$ }
    remove_from_variable_map = { name = cmf_alert_open_view key = $target$ }
    remove_from_variable_map = { name = cmf_alert_open_group key = $target$ }
    remove_from_variable_map = { name = cmf_alert_open_window key = $target$ }
}
