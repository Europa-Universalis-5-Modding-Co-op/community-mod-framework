# Suppress "used but never set" and "set but never used" engine warnings for a name.
# Scope: any
# Args: v (key)
cmf_suppress = {
	if = {
		limit = {
			always = no
			exists = var:$v$
			exists = flag:$v$
			exists = scope:$v$
			exists = global_var:$v$
			has_variable_list = $v$
			has_global_variable_list = $v$
			has_variable_map = $v$
			has_global_variable_map = $v$
		}
		set_variable = { name = $v$ value = flag:$v$ }
		save_scope_as = $v$
		set_global_variable = { name = $v$ value = flag:$v$ }
		add_to_variable_list = { name = $v$ target = flag:$v$ }
		add_to_global_variable_list = { name = $v$ target = flag:$v$ }
		add_to_variable_map = { name = $v$ key = flag:$v$ value = flag:$v$ }
		add_to_global_variable_map = { name = $v$ key = flag:$v$ value = flag:$v$ }
	}
}

# DEPRECATED: add_to_variable_map now overwrites an existing key; call it directly instead.
# Set a variable map entry, replacing any existing value
# Scope: any
# Args: name (key), key (scope), value (scope)
cmf_change_variable_map = {
	remove_from_variable_map = {
		name = $name$
		key = $key$
	}
	add_to_variable_map = {
		name = $name$
		key = $key$
		value = $value$
	}
}
