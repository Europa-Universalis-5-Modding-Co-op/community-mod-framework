# Register built-in CMM settings.
# Scope: country
# Args: none
_cmm_register_menu_settings = {
	if = {
		limit = { has_multiple_players = no }
		cmm_register_global_bool_setting = {
			mod_id = cmm_core
			setting_id = enable_host_only_tools
			tab_id = general
			group_id = session
			default_value = 1
		}
	}
	else = {
		cmm_register_global_bool_setting = {
			mod_id = cmm_core
			setting_id = enable_host_only_tools
			tab_id = general
			group_id = session
			default_value = 0
		}
	}

	# Debug mode is singleplayer-only (console command does not work in multiplayer)
	if = {
		limit = { has_multiple_players = no }
		cmm_register_global_bool_setting = {
			mod_id = cmm_core
			setting_id = enable_debug_mode
			tab_id = general
			group_id = session
			default_value = 0
		}
	}

	cmm_register_button_setting = {
		mod_id = cmm_core
		setting_id = open_log
		tab_id = general
		group_id = session
	}
}

# Handle CMM core button callbacks.
# Scope: country
# Args: none
_cmm_handle_core_callback = {
	if = {
		limit = { var:cmf_callback = flag:cmm_core__open_log }
		if = {
			limit = { has_global_variable = cmf_log_view_open }
			remove_global_variable = cmf_log_view_open
		}
		else = {
			set_global_variable = { name = cmf_log_view_open value = yes }
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmf_log_view_open }
	}
}

# Store the current country as the CMM host country.
# Scope: country
# Args: none
_cmm_mark_host_country = {
	set_global_variable = { name = cmm_core_host_country value = this }

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_core_host_country }
	}
}
