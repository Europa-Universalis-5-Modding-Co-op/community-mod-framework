# Reserve one ordered list field slot for a field id.
# Scope: country
# Args: mod_id (key), setting_id (key), field_id (key)
_cmm_register_list_field_slot = {
	if = {
		limit = { NOT = { has_variable = cmm_list_field_count_$mod_id$__$setting_id$ } }
		set_variable = { name = cmm_list_field_count_$mod_id$__$setting_id$ value = 0 }
	}

	if = {
		limit = {
			AND = {
				NOT = { has_variable = cmm_registered_list_field_$mod_id$__$setting_id$__$field_id$ }
				var:cmm_list_field_count_$mod_id$__$setting_id$ < 5
			}
		}
		set_variable = { name = cmm_registered_list_field_$mod_id$__$setting_id$__$field_id$ value = 1 }
		change_variable = { name = cmm_list_field_count_$mod_id$__$setting_id$ add = 1 }
		set_variable = {
			name = cmm_list_field_slot_$mod_id$__$setting_id$__$field_id$
			value = var:cmm_list_field_count_$mod_id$__$setting_id$
		}
	}
}

# Apply a list field slot's localization and format from the persistent override
# stores, or the field-id defaults when none is recorded.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Owner setting id.
# - field_id (key): Field id unique within the list setting.
# - slot (number): Field slot index (`1..5`).
_cmm_apply_list_field_loc_format = {
	set_local_variable = { name = cmm_lff_key value = flag:$mod_id$__$setting_id$__$field_id$ }

	if = {
		limit = { is_key_in_variable_map = { name = cmm_list_field_format_pending target = local_var:cmm_lff_key } }
		set_local_variable = { name = cmm_lff_format value = "variable_map(cmm_list_field_format_pending|local_var:cmm_lff_key)" }
		set_variable = { name = cmm_list_field_$slot$_has_format_$mod_id$__$setting_id$ value = local_var:cmm_lff_format }
	}
	else = {
		set_variable = { name = cmm_list_field_$slot$_has_format_$mod_id$__$setting_id$ value = 0 }
	}

	if = {
		limit = { is_key_in_variable_map = { name = cmm_list_field_loc_name_pending target = local_var:cmm_lff_key } }
		set_local_variable = { name = cmm_lff_name value = "variable_map(cmm_list_field_loc_name_pending|local_var:cmm_lff_key)" }
		set_variable = { name = cmm_list_field_$slot$_name_$mod_id$__$setting_id$ value = local_var:cmm_lff_name }
	}
	else = {
		set_variable = { name = cmm_list_field_$slot$_name_$mod_id$__$setting_id$ value = flag:$mod_id$__$setting_id$__$field_id$_name }
	}

	if = {
		limit = { is_key_in_variable_map = { name = cmm_list_field_loc_key_pending target = local_var:cmm_lff_key } }
		set_local_variable = { name = cmm_lff_root value = "variable_map(cmm_list_field_loc_key_pending|local_var:cmm_lff_key)" }
		set_variable = { name = cmm_list_field_$slot$_key_$mod_id$__$setting_id$ value = local_var:cmm_lff_root }
	}
	else = {
		set_variable = { name = cmm_list_field_$slot$_key_$mod_id$__$setting_id$ value = flag:$mod_id$__$setting_id$__$field_id$ }
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_lff_key }
		cmf_suppress = { v = cmm_lff_format }
		cmf_suppress = { v = cmm_lff_name }
		cmf_suppress = { v = cmm_lff_root }
	}
}

	set_local_variable = { name = cmm_eff_default value = $default_value$ }

		limit = { is_key_in_variable_map = { name = cmm_list_field_default_pending target = flag:$mod_id$__$setting_id$__$field_id$_i$item$ } }

		set_local_variable = { name = cmm_eff_default value = "variable_map(cmm_list_field_default_pending|local_var:cmm_pfd_key)" }
	}

	add_to_variable_map = { name = cmm_is_global key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = $is_global$ }
	add_to_variable_map = { name = cmm_list_field_default key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = local_var:cmm_eff_default }
	add_to_variable_map = { name = cmm_list_field_owner key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = flag:$mod_id$__$setting_id$ }
	if = {
		limit = { NOT = { is_key_in_$vt$_map = { name = cmm target = flag:$mod_id$__$setting_id$_i$item$_f$slot$ } } }
		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = local_var:cmm_eff_default }
		add_to_variable_list = { name = cmm_list_field_seeded_this_pass target = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
	}

	set_local_variable = { name = cmm_tmp_fk value = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
	if = {
		limit = { $is_global$ >= 1 }
		set_local_variable = { name = cmm_field_val value = "global_variable_map(cmm|local_var:cmm_tmp_fk)" }
	}
	else = {
		set_local_variable = { name = cmm_field_val value = "variable_map(cmm|local_var:cmm_tmp_fk)" }
	}

	if = {
		limit = { local_var:cmm_field_val < 0 }
		remove_from_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = 0 }

		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = 1 }
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_tmp_fk }
		cmf_suppress = { v = cmm_field_val }
		cmf_suppress = { v = cmm_pfd_key }
	}

# Initialize one ordered list dropdown field value for one stable item identity.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Owner setting id.
# - field_id (key): Field id unique within the list setting.
# - slot (number): Field slot index (`1..5`).
# - item (number): Stable item identity ordinal (`1..50`).
# - default_index (number): Default selected option ordinal for new saves.
# - vt (variable/global_variable): Variable type for map commands.
# - is_global (1/0): `1` for global variable storage, `0` for country variable storage.
_cmm_initialize_list_dropdown_field_item = {
	set_local_variable = { name = cmm_eff_default value = $default_index$ }
	if = {
		limit = { is_key_in_variable_map = { name = cmm_list_field_default_pending target = flag:$mod_id$__$setting_id$__$field_id$_i$item$ } }
		set_local_variable = { name = cmm_pfd_key value = flag:$mod_id$__$setting_id$__$field_id$_i$item$ }
		set_local_variable = { name = cmm_eff_default value = "variable_map(cmm_list_field_default_pending|local_var:cmm_pfd_key)" }
	}

	add_to_variable_map = { name = cmm_is_global key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = $is_global$ }
	add_to_variable_map = { name = cmm_list_field_default key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = local_var:cmm_eff_default }
	add_to_variable_map = { name = cmm_list_field_owner key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = flag:$mod_id$__$setting_id$ }
	if = {
		limit = { NOT = { is_key_in_$vt$_map = { name = cmm target = flag:$mod_id$__$setting_id$_i$item$_f$slot$ } } }
		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = local_var:cmm_eff_default }
		add_to_variable_list = { name = cmm_list_field_seeded_this_pass target = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
	}

	set_local_variable = { name = cmm_tmp_fk value = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
	if = {
		limit = { $is_global$ >= 1 }
		set_local_variable = { name = cmm_field_val value = "global_variable_map(cmm|local_var:cmm_tmp_fk)" }
	}
	else = {
		set_local_variable = { name = cmm_field_val value = "variable_map(cmm|local_var:cmm_tmp_fk)" }
	}

	if = {
		limit = { local_var:cmm_field_val < 1 }
		remove_from_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = 1 }
	}

	if = {
		limit = { local_var:cmm_field_val > var:cmm_list_field_$slot$_dropdown_last_index_$mod_id$__$setting_id$ }
		remove_from_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
		add_to_$vt$_map = { name = cmm key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = var:cmm_list_field_$slot$_dropdown_last_index_$mod_id$__$setting_id$ }
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_tmp_fk }
		cmf_suppress = { v = cmm_field_val }
		cmf_suppress = { v = cmm_pfd_key }
	}
}

	}

	add_to_variable_map = { name = cmm_list_field_owner key = flag:$mod_id$__$setting_id$_i$item$_f$slot$ value = flag:$mod_id$__$setting_id$ }
	if = {
		limit = { NOT = { is_key_in_$vt$_map = { name = cmm target = flag:$mod_id$__$setting_id$_i$item$_f$slot$ } } }

	}

	set_local_variable = { name = cmm_tmp_fk value = flag:$mod_id$__$setting_id$_i$item$_f$slot$ }
	if = {
		limit = { $is_global$ >= 1 }
		set_local_variable = { name = cmm_field_val value = "global_variable_map(cmm|local_var:cmm_tmp_fk)" }
	}
	else = {

	}

	if = {
		limit = { local_var:cmm_field_val < var:cmm_list_field_$slot$_min_$mod_id$__$setting_id$ }
