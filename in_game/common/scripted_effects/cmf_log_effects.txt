# Log an action to the global mod log.
# Scope: country (the actor performing the action)
# Args: action (flag) — localization key for the action text
cmf_log = {
	save_scope_as = _cmf_log_actor

	if = {
		limit = { NOT = { has_global_variable = cmf_log_count } }
		set_global_variable = { name = cmf_log_count value = 0 }
	}

	if = {
		limit = { global_var:cmf_log_count < 50 }
		_cmf_log_dispatch = { action = $action$ }
		add_to_global_variable_list = { name = cmf_log_entries value = scope:_cmf_log_actor }
		change_global_variable = { name = cmf_log_count add = 1 }
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = _cmf_log_actor }
	}
}

# Clear all log entries.
# Scope: any
# Args: none
cmf_clear_log = {
	clear_global_variable_list = cmf_log_entries
	set_global_variable = { name = cmf_log_count value = 0 }
}

# Internal: dispatch to the correct slot group based on current count.
# Scope: country
# Args: action (flag)
_cmf_log_dispatch = {
	if = {
		limit = { global_var:cmf_log_count < 10 }
		_cmf_log_write_0x = { action = $action$ }
	}
	else_if = {
		limit = { global_var:cmf_log_count < 20 }
		_cmf_log_write_1x = { action = $action$ }
	}
	else_if = {
		limit = { global_var:cmf_log_count < 30 }
		_cmf_log_write_2x = { action = $action$ }
	}
	else_if = {
		limit = { global_var:cmf_log_count < 40 }
		_cmf_log_write_3x = { action = $action$ }
	}
	else = {
		_cmf_log_write_4x = { action = $action$ }
	}
}
