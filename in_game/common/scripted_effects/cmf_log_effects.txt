# Log an action to the global mod log.
# Scope: country (the actor performing the action)
# Args: action (flag) — localization key for the action text
cmf_log = {
	_cmf_log_core = { action = $action$ arg1 = _cmf_log_no_arg arg2 = _cmf_log_no_arg }
}

# Log an action with two named arguments.
# Entry text renders as: [arg1] [action] [arg2]
# Scope: country (the actor performing the action)
# Args:
# - action (flag): localization key for the action/connecting text
# - arg1 (flag): localization key for the first argument (before action text)
# - arg2 (flag): localization key for the second argument (after action text)
cmf_log_with_args = {
	_cmf_log_core = { action = $action$ arg1 = $arg1$ arg2 = $arg2$ }
}

# Clear all log entries.
# Scope: country
# Args: none
cmf_clear_log = {
	while = {
		limit = { global_var:cmf_log_count > 0 }
		set_variable = { name = _cmf_log_tmp value = global_var:cmf_log_count }
		change_variable = { name = _cmf_log_tmp subtract = 1 }
		set_global_variable = { name = cmf_log_count value = var:_cmf_log_tmp }
		remove_from_global_variable_map = { name = cmf_log_action key = var:_cmf_log_tmp }
		remove_from_global_variable_map = { name = cmf_log_actor key = var:_cmf_log_tmp }
		remove_from_global_variable_map = { name = cmf_log_arg1 key = var:_cmf_log_tmp }
		remove_from_global_variable_map = { name = cmf_log_arg2 key = var:_cmf_log_tmp }
		remove_variable = _cmf_log_tmp
	}
}

# Internal: shared log core.
# Scope: country (the actor)
# Args: action (flag), arg1 (flag), arg2 (flag)
_cmf_log_core = {
	save_scope_as = _cmf_log_actor

	if = {
		limit = { NOT = { has_global_variable = cmf_log_count } }
		set_global_variable = { name = cmf_log_count value = 0 }
	}

	# Use current count as the map key
	set_variable = { name = _cmf_log_idx value = global_var:cmf_log_count }

	add_to_global_variable_map = { name = cmf_log_action key = var:_cmf_log_idx value = flag:$action$ }
	add_to_global_variable_map = { name = cmf_log_actor key = var:_cmf_log_idx value = scope:_cmf_log_actor }
	add_to_global_variable_map = { name = cmf_log_arg1 key = var:_cmf_log_idx value = flag:$arg1$ }
	add_to_global_variable_map = { name = cmf_log_arg2 key = var:_cmf_log_idx value = flag:$arg2$ }

	# Increment count
	change_variable = { name = _cmf_log_idx add = 1 }
	set_global_variable = { name = cmf_log_count value = var:_cmf_log_idx }
	remove_variable = _cmf_log_idx

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = _cmf_log_actor }
		cmf_suppress = { v = _cmf_log_no_arg }
		cmf_suppress = { v = _cmf_log_idx }
		cmf_suppress = { v = _cmf_log_tmp }
	}
}
