# Rebuild one ordered list after moving the marked row to `local_var:cmm_list_target_index`.
# Scope: country
# Args:
# - setting (key): Composite setting key `<mod_id>__<setting_id>`.
# - vt (variable/global_variable): Variable type token.
# - lt (in/in_global): List iteration token.
_cmm_reorder_list_setting = {
	clear_variable_list = cmm_list_rebuild_temp
	set_local_variable = { name = cmm_list_source_index value = 0 }
	set_local_variable = { name = cmm_list_found_moving_item value = 0 }
	set_local_variable = { name = cmm_list_marked_position value = var:cmm_list_position_to_apply }

	every_$lt$_list = {
		variable = cmm_list_items_$setting$
		save_scope_as = cmm_list_current_item

		root = {
			add_to_variable_list = {
				name = cmm_list_rebuild_temp
				target = scope:cmm_list_current_item
			}
		}

		if = {
			limit = { local_var:cmm_list_source_index = local_var:cmm_list_marked_position }
			save_scope_as = cmm_list_moving_item
			set_local_variable = { name = cmm_list_found_moving_item value = 1 }
		}

		change_local_variable = { name = cmm_list_source_index add = 1 }
	}

	if = {
		limit = { local_var:cmm_list_found_moving_item = 1 }

		clear_$vt$_list = cmm_list_items_$setting$

		set_local_variable = { name = cmm_list_reinsert_index value = 0 }
		set_local_variable = { name = cmm_list_inserted_moving_item value = 0 }

		every_in_list = {
			variable = cmm_list_rebuild_temp
			save_scope_as = cmm_list_current_item

			if = {
				limit = { NOT = { this = scope:cmm_list_moving_item } }
				if = {
					limit = {
						AND = {
							local_var:cmm_list_reinsert_index = local_var:cmm_list_target_index
							local_var:cmm_list_inserted_moving_item = 0
						}
					}
					root = {
						add_to_$vt$_list = {
							name = cmm_list_items_$setting$
							target = scope:cmm_list_moving_item
						}
					}
					set_local_variable = { name = cmm_list_inserted_moving_item value = 1 }
				}

				root = {
					add_to_$vt$_list = {
						name = cmm_list_items_$setting$
						target = scope:cmm_list_current_item
					}
				}
				change_local_variable = { name = cmm_list_reinsert_index add = 1 }
			}
		}

		if = {
			limit = { local_var:cmm_list_inserted_moving_item = 0 }
			add_to_$vt$_list = {
				name = cmm_list_items_$setting$
				target = scope:cmm_list_moving_item
			}
		}
	}

	clear_variable_list = cmm_list_rebuild_temp
}

# Apply the queued numeric markers to one ordered list numeric field slot for one item identity.
# Scope: country
# Args:
# - setting (key): Composite setting key `<mod_id>__<setting_id>`.
# - item (number): Stable item identity ordinal (`1..50`).
# - slot (number): Field slot index (`1..5`).
# - vt (variable/global_variable): Variable type token.
_cmm_list_apply_numeric_field_slot = {
	set_local_variable = { name = cmm_tmp_disabled_fk value = flag:$setting$_i$item$_f$slot$ }
	if = {
		limit = { NOT = { is_key_in_variable_map = { name = cmm_list_field_disabled target = local_var:cmm_tmp_disabled_fk } } }
		if = {
			limit = { AND = { OR = { var:cmm_list_field_$slot$_type_$setting$ = 3 var:cmm_list_field_$slot$_type_$setting$ = 4 } has_variable = cmm_numeric_step_delta_to_apply } }
			set_local_variable = { name = cmm_tmp_fk value = flag:$setting$_i$item$_f$slot$ }
			if = {
				limit = { var:cmm_tmp_is_global_list >= 1 }
				set_variable = { name = cmm_tmp_numeric_value value = "global_variable_map(cmm|local_var:cmm_tmp_fk)" }
			}
			else = {
				set_variable = { name = cmm_tmp_numeric_value value = "variable_map(cmm|local_var:cmm_tmp_fk)" }
			}
			set_variable = { name = cmm_tmp_numeric_delta value = var:cmm_list_field_$slot$_step_$setting$ }
			change_variable = { name = cmm_tmp_numeric_delta multiply = var:cmm_numeric_step_delta_to_apply }
			change_variable = { name = cmm_tmp_numeric_value add = var:cmm_tmp_numeric_delta }

			if = {
				limit = { var:cmm_tmp_numeric_value > var:cmm_list_field_$slot$_max_$setting$ }
				set_variable = { name = cmm_tmp_numeric_value value = var:cmm_list_field_$slot$_max_$setting$ }
			}

			if = {
				limit = { var:cmm_tmp_numeric_value < var:cmm_list_field_$slot$_min_$setting$ }
				set_variable = { name = cmm_tmp_numeric_value value = var:cmm_list_field_$slot$_min_$setting$ }
			}

			remove_from_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ }
			add_to_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ value = var:cmm_tmp_numeric_value }
			remove_variable = cmm_tmp_numeric_value
			remove_variable = cmm_tmp_numeric_delta
		}
		else_if = {
			limit = { AND = { OR = { var:cmm_list_field_$slot$_type_$setting$ = 3 var:cmm_list_field_$slot$_type_$setting$ = 4 } has_variable = cmm_numeric_change_to_apply } }
			if = {
				limit = { var:cmm_numeric_change_to_apply > 0 }
				remove_from_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ }
				add_to_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ value = var:cmm_list_field_$slot$_max_$setting$ }
			}
			else_if = {
				limit = { var:cmm_numeric_change_to_apply < 0 }
				remove_from_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ }
				add_to_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ value = var:cmm_list_field_$slot$_min_$setting$ }
			}
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_tmp_fk }
		cmf_suppress = { v = cmm_tmp_disabled_fk }
		cmf_suppress = { v = cmm_list_field_disabled }
	}
}

				limit = { AND = { var:cmm_list_field_$slot$_type_$setting$ = 1 var:cmm_tmp_field_new > 1 } }
				set_variable = { name = cmm_tmp_field_new value = 1 }
			}
			if = {
				limit = { AND = { var:cmm_list_field_$slot$_type_$setting$ = 2 var:cmm_tmp_field_new < 1 } }
				set_variable = { name = cmm_tmp_field_new value = 1 }
			}
			if = {
				limit = { AND = { var:cmm_list_field_$slot$_type_$setting$ = 2 var:cmm_tmp_field_new > var:cmm_list_field_$slot$_dropdown_last_index_$setting$ } }
				set_variable = { name = cmm_tmp_field_new value = var:cmm_list_field_$slot$_dropdown_last_index_$setting$ }
			}
			if = {
				limit = { AND = { OR = { var:cmm_list_field_$slot$_type_$setting$ = 3 var:cmm_list_field_$slot$_type_$setting$ = 4 } var:cmm_tmp_field_new < var:cmm_list_field_$slot$_min_$setting$ } }
				set_variable = { name = cmm_tmp_field_new value = var:cmm_list_field_$slot$_min_$setting$ }
			}
			if = {
				limit = { AND = { OR = { var:cmm_list_field_$slot$_type_$setting$ = 3 var:cmm_list_field_$slot$_type_$setting$ = 4 } var:cmm_tmp_field_new > var:cmm_list_field_$slot$_max_$setting$ } }
				set_variable = { name = cmm_tmp_field_new value = var:cmm_list_field_$slot$_max_$setting$ }
			}
			remove_from_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ }
			add_to_$vt$_map = { name = cmm key = flag:$setting$_i$item$_f$slot$ value = var:cmm_tmp_field_new }
			remove_variable = cmm_tmp_field_new
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_tmp_fk }
		cmf_suppress = { v = cmm_field_val }
		cmf_suppress = { v = cmm_tmp_disabled_fk }
	}
}

# Iterate over every item in a list setting and call a user-provided effect for each.
# The effect receives `$i$` as the resolved item number (1-50) and `$setting$` as the
# composite key. A macro param does not resolve inside a quoted variable_map lookup, so
# build the field key into a local first to read a field value:
#   set_local_variable = { name = cmm_tmp_fk value = flag:$setting$_i$i$_f1 }
#   "variable_map(cmm|local_var:cmm_tmp_fk)"
# If the item has a value (set via cmm_set_list_item_value or cmm_add_settings_list_item),
# it is available as `scope:cmm_list_current_item_value` inside the called effect.
# Scope: country
# Args:
# - setting (key): Composite setting key `<mod_id>__<setting_id>`.
# - effect (key): Name of the scripted effect to call per item.
cmm_for_each_list_item = {
	set_local_variable = { name = cmm_tmp_flag value = flag:$setting$ }
	if = {
		limit = { "variable_map(cmm_is_global|local_var:cmm_tmp_flag)" >= 1 }
		every_in_global_list = {
			variable = cmm_list_items_$setting$
			save_scope_as = cmm_list_current_item

			root = {
				_cmm_resolve_list_item_and_call = { setting = $setting$ effect = $effect$ }
			}
		}
	}
	else = {
		every_in_list = {
			variable = cmm_list_items_$setting$
			save_scope_as = cmm_list_current_item

			root = {
				_cmm_resolve_list_item_and_call = { setting = $setting$ effect = $effect$ }
			}
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_tmp_flag }
	}
}

# Build a variable list of item values in display order (respects list reordering).
# Requires item values set via cmm_set_list_item_value. Clears the target list first.
# Scope: country
# Args:
# - setting (key): Composite setting key `<mod_id>__<setting_id>`.
# - list_name (key): Name of the variable list to populate.
cmm_build_list_ordered_values = {
	clear_variable_list = $list_name$
	clear_variable_list = _cmm_tmp_ordered_values
	cmm_for_each_list_item = {
		setting = $setting$
		effect = _cmm_build_list_ordered_values_item
	}
	every_in_list = {
		variable = _cmm_tmp_ordered_values
		save_scope_as = _cmm_tmp_val
		root = { add_to_variable_list = { name = $list_name$ target = scope:_cmm_tmp_val } }
	}
	clear_variable_list = _cmm_tmp_ordered_values

	if = {
		limit = { always = no }
		cmf_suppress = { v = _cmm_tmp_ordered_values }
		cmf_suppress = { v = _cmm_tmp_val }
	}
}

# Internal: called per item by cmm_build_list_ordered_values.
# Writes to fixed temp list; parent copies to target after iteration.
# Scope: country
# Args: setting (key), i (number)
_cmm_build_list_ordered_values_item = {
	add_to_variable_list = { name = _cmm_tmp_ordered_values target = scope:cmm_list_current_item_value }

	# Accept setting/i params from cmm_for_each_list_item (unused here)
	if = {
		limit = { always = no }
		cmf_suppress = { v = $setting$_$i$ }
	}
}

# Build a variable map from a list field, keyed by item values (game object scopes).
# Requires item values set via cmm_set_list_item_value. Clears the target map first.
# Scope: country
# Args:
# - setting (key): Composite setting key `<mod_id>__<setting_id>`.
# - field_slot (number): Field slot number (1-5).
# - map_name (key): Name of the variable map to populate.
# - multiply (number): Scale factor applied to each value (e.g. 0.01 to convert 0-100 to 0-1). Use 1 for no scaling.
cmm_build_list_field_map = {
	clear_variable_map = $map_name$
	clear_variable_map = _cmm_tmp_field_map
	set_variable = { name = cmm_blfm_field_slot value = $field_slot$ }
	set_variable = { name = cmm_blfm_multiply value = $multiply$ }
	set_local_variable = { name = cmm_tmp_flag value = flag:$setting$ }
	set_variable = { name = cmm_blfm_is_global value = "variable_map(cmm_is_global|local_var:cmm_tmp_flag)" }
	cmm_for_each_list_item = {
		setting = $setting$
		effect = _cmm_build_list_field_map_item
	}
	every_key_in_variable_map = {
		variable = _cmm_tmp_field_map
		save_scope_as = _cmm_tmp_map_key
		root = {
			set_local_variable = { name = _cmm_tmp_map_val value = "variable_map(_cmm_tmp_field_map|scope:_cmm_tmp_map_key)" }
			add_to_variable_map = { name = $map_name$ key = scope:_cmm_tmp_map_key value = local_var:_cmm_tmp_map_val }
		}
	}
	clear_variable_map = _cmm_tmp_field_map

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_blfm_field_slot }
		cmf_suppress = { v = cmm_blfm_multiply }
		cmf_suppress = { v = cmm_blfm_is_global }
		cmf_suppress = { v = cmm_blfm_value }
		cmf_suppress = { v = _cmm_tmp_field_map }
		cmf_suppress = { v = _cmm_tmp_map_key }
		cmf_suppress = { v = cmm_tmp_key }
	}
}

# Internal: called per item by cmm_build_list_field_map.
# Writes to fixed temp map; parent copies to target after iteration.
# Scope: country
# Args: setting (key), i (number)
_cmm_build_list_field_map_item = {
	set_local_variable = { name = cmm_tmp_key value = flag:$setting$_i$i$_f1 }
	if = { limit = { var:cmm_blfm_field_slot = 2 } set_local_variable = { name = cmm_tmp_key value = flag:$setting$_i$i$_f2 } }
	if = { limit = { var:cmm_blfm_field_slot = 3 } set_local_variable = { name = cmm_tmp_key value = flag:$setting$_i$i$_f3 } }
	if = { limit = { var:cmm_blfm_field_slot = 4 } set_local_variable = { name = cmm_tmp_key value = flag:$setting$_i$i$_f4 } }
	if = { limit = { var:cmm_blfm_field_slot = 5 } set_local_variable = { name = cmm_tmp_key value = flag:$setting$_i$i$_f5 } }

	if = {
		limit = { var:cmm_blfm_is_global >= 1 }
		set_variable = { name = cmm_blfm_value value = "global_variable_map(cmm|local_var:cmm_tmp_key)" }
	}
	else = {
		set_variable = { name = cmm_blfm_value value = "variable_map(cmm|local_var:cmm_tmp_key)" }
	}
	change_variable = { name = cmm_blfm_value multiply = var:cmm_blfm_multiply }
	add_to_variable_map = { name = _cmm_tmp_field_map key = scope:cmm_list_current_item_value value = var:cmm_blfm_value }
}

# Internal: save the item's value scope (if set) and invoke the user-provided effect.
# The value is read from the single-element variable list `<setting>_item_<item>_value`
# and exposed as `scope:cmm_list_current_item_value` inside the called effect.
# If no value was set for the item, the scope is simply not available.
# Scope: country
# Args: setting (key), item (number), effect (key)
_cmm_save_list_item_value_and_call = {
	every_in_list = {
		variable = $setting$_i$item$_value
		save_scope_as = cmm_list_current_item_value
	}
	$effect$ = { setting = $setting$ i = $item$ }
}
