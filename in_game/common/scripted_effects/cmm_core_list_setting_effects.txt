# Register a settings list for a mod (idempotent by mod_id + setting_id).
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Setting id unique within mod.
# - tab_id (key): Owner tab id within mod.
# - item_count (number): Number of list items (`1..20`).
# - is_ordered (1/0): `1` shows row move controls, `0` hides them.
cmm_register_settings_list = {
	cmm_register_list_setting_internal = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		tab_id = $tab_id$
		item_count = $item_count$
		is_ordered = $is_ordered$
		is_global = 0
		vt = variable
	}
}

# Register shared list-setting metadata and storage for a mod (idempotent by mod_id + setting_id).
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Setting id unique within mod.
# - tab_id (key): Owner tab id within mod.
# - item_count (number): Number of list items (`1..20`).
# - is_ordered (1/0): `1` shows row move controls, `0` hides them.
# - is_global (1/0): `1` for global variable storage, `0` for country variable storage.
# - vt (variable/global_variable): Variable type token for macro composition.
cmm_register_list_setting_internal = {
	add_to_variable_map = { name = cmm_type key = flag:$mod_id$__$setting_id$ value = 7 }
	add_to_variable_map = { name = cmm_is_global key = flag:$mod_id$__$setting_id$ value = $is_global$ }
	set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = $is_ordered$ }
	set_variable = { name = cmm_setting_list_count_$mod_id$__$setting_id$ value = $item_count$ }

	if = {
		limit = { var:cmm_setting_list_is_ordered_$mod_id$__$setting_id$ > 0 }
		set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = 1 }
	}
	else = {
		set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = 0 }
	}

	if = {
		limit = { var:cmm_setting_list_count_$mod_id$__$setting_id$ < 1 }
		set_variable = { name = cmm_setting_list_count_$mod_id$__$setting_id$ value = 1 }
	}

	if = {
		limit = { var:cmm_setting_list_count_$mod_id$__$setting_id$ > 20 }
		set_variable = { name = cmm_setting_list_count_$mod_id$__$setting_id$ value = 20 }
	}

	set_variable = {
		name = cmm_setting_list_last_index_$mod_id$__$setting_id$
		value = var:cmm_setting_list_count_$mod_id$__$setting_id$
	}
	change_variable = { name = cmm_setting_list_last_index_$mod_id$__$setting_id$ add = -1 }

	add_to_variable_map = { name = cmm_list_count key = flag:$mod_id$__$setting_id$ value = var:cmm_setting_list_count_$mod_id$__$setting_id$ }
	add_to_variable_map = { name = cmm_list_is_ordered key = flag:$mod_id$__$setting_id$ value = var:cmm_setting_list_is_ordered_$mod_id$__$setting_id$ }
	add_to_variable_map = { name = cmm_list_last_index key = flag:$mod_id$__$setting_id$ value = var:cmm_setting_list_last_index_$mod_id$__$setting_id$ }

	if = {
		limit = { NOT = { has_variable = cmm_list_field_count_$mod_id$__$setting_id$ } }
		set_variable = { name = cmm_list_field_count_$mod_id$__$setting_id$ value = 0 }
	}

	set_variable = {
		name = cmm_list_item_column_name_$mod_id$__$setting_id$
		value = flag:$mod_id$__$setting_id$_item_column_name
	}

	cmm_register_setting_metadata = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		tab_id = $tab_id$
		group_id = $setting_id$
	}

	cmm_initialize_list_setting_item_metadata = {
		mod_id = $mod_id$
		setting_id = $setting_id$
	}

	if = {
		limit = { NOT = { has_variable = cmm_list_initialized_$mod_id$__$setting_id$ } }

		cmm_initialize_list_setting_item_list = {
			mod_id = $mod_id$
			setting_id = $setting_id$
			vt = $vt$
		}

		set_variable = { name = cmm_list_initialized_$mod_id$__$setting_id$ value = 1 }
	}

	cmm_suppress_list_setting_warnings = {
		mod_id = $mod_id$
		setting_id = $setting_id$
	}
}

# Register a bool field shared by every item in one list setting.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Target list setting id within the mod.
# - field_id (key): Field id unique within the list setting.
# - default_value (1/0): Default checkbox state for new saves.
cmm_register_list_bool_field = {
	cmm_register_list_field_slot = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		field_id = $field_id$
	}

	set_local_variable = { name = cmm_tmp_flag value = flag:$mod_id$__$setting_id$ }
	set_local_variable = { name = cmm_tmp_is_global value = "variable_map(cmm_is_global|local_var:cmm_tmp_flag)" }

	if = {
		limit = {
			AND = {
				has_variable = cmm_setting_list_count_$mod_id$__$setting_id$
				has_variable = cmm_list_field_slot_$mod_id$__$setting_id$__$field_id$
			}
		}
		if = {
			limit = { local_var:cmm_tmp_is_global >= 1 }
			switch = {
				trigger = var:cmm_list_field_slot_$mod_id$__$setting_id$__$field_id$
				1 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 1 default_value = $default_value$ vt = global_variable is_global = 1 } }
				2 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 2 default_value = $default_value$ vt = global_variable is_global = 1 } }
				3 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 3 default_value = $default_value$ vt = global_variable is_global = 1 } }
				4 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 4 default_value = $default_value$ vt = global_variable is_global = 1 } }
				5 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 5 default_value = $default_value$ vt = global_variable is_global = 1 } }
			}
		}
		else = {
			switch = {
				trigger = var:cmm_list_field_slot_$mod_id$__$setting_id$__$field_id$
				1 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 1 default_value = $default_value$ vt = variable is_global = 0 } }
				2 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 2 default_value = $default_value$ vt = variable is_global = 0 } }
				3 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 3 default_value = $default_value$ vt = variable is_global = 0 } }
				4 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 4 default_value = $default_value$ vt = variable is_global = 0 } }
				5 = { cmm_initialize_list_bool_field_slot = { mod_id = $mod_id$ setting_id = $setting_id$ field_id = $field_id$ slot = 5 default_value = $default_value$ vt = variable is_global = 0 } }
			}
		}
	}

	cmm_suppress_list_bool_field_warnings = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		field_id = $field_id$
	}

	# Benign warning suppression
	if = {
		limit = {
			always = no
			exists = local_var:cmm_tmp_flag
			exists = local_var:cmm_tmp_is_global
		}
	}
}

# Attach a game object scope as the value of one list item (call after cmm_register_settings_list).
# Stores the scope in a single-element variable list per item.
# When iterating with cmm_for_each_list_item, the scope is available as
# `scope:cmm_list_current_item_value` inside your per-item effect.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Target list setting id within the mod.
# - item (number): Item ordinal to tag (`1..item_count`).
# - value (scope): Game scope to attach (e.g. `building_type:fine_cloth_guild`).
cmm_set_list_item_value = {
	clear_variable_list = $mod_id$__$setting_id$_item_$item$_value
	add_to_variable_list = { name = $mod_id$__$setting_id$_item_$item$_value target = $value$ }
}

# Begin a dynamic settings list whose items are added one by one via cmm_add_settings_list_item
# or built automatically from a variable list via cmm_register_settings_list_from_list.
# Must be followed by item-add calls and then cmm_finish_settings_list.
# Field registration (cmm_register_list_bool_field, etc.) must come after cmm_finish_settings_list.
# Idempotent: subsequent calls refresh metadata without re-adding items.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Setting id unique within mod.
# - tab_id (key): Owner tab id within mod.
# - is_ordered (1/0): `1` shows row move controls, `0` hides them.
cmm_begin_settings_list = {
	add_to_variable_map = { name = cmm_type key = flag:$mod_id$__$setting_id$ value = 7 }
	add_to_variable_map = { name = cmm_is_global key = flag:$mod_id$__$setting_id$ value = 0 }
	set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = $is_ordered$ }

	if = {
		limit = { var:cmm_setting_list_is_ordered_$mod_id$__$setting_id$ > 0 }
		set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = 1 }
	}
	else = {
		set_variable = { name = cmm_setting_list_is_ordered_$mod_id$__$setting_id$ value = 0 }
	}

	add_to_variable_map = { name = cmm_list_is_ordered key = flag:$mod_id$__$setting_id$ value = var:cmm_setting_list_is_ordered_$mod_id$__$setting_id$ }

	if = {
		limit = { NOT = { has_variable = cmm_list_field_count_$mod_id$__$setting_id$ } }
		set_variable = { name = cmm_list_field_count_$mod_id$__$setting_id$ value = 0 }
	}

	set_variable = {
		name = cmm_list_item_column_name_$mod_id$__$setting_id$
		value = flag:$mod_id$__$setting_id$_item_column_name
	}

	cmm_register_setting_metadata = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		tab_id = $tab_id$
		group_id = $setting_id$
	}

	if = {
		limit = { NOT = { has_variable = cmm_list_initialized_$mod_id$__$setting_id$ } }
		set_variable = { name = cmm_list_add_counter_$mod_id$__$setting_id$ value = 0 }
		clear_variable_list = cmm_list_items_$mod_id$__$setting_id$
	}
	else = {
		cmm_reinitialize_dynamic_list_item_metadata = { mod_id = $mod_id$ setting_id = $setting_id$ }
	}
}

# Finalize a dynamic settings list after all item-add calls.
# Sets the item count from the accumulated add counter and marks the list initialized.
# Register fields (cmm_register_list_bool_field, etc.) after this call.
# Idempotent: subsequent calls are no-ops.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Target list setting id within the mod.
cmm_finish_settings_list = {
	if = {
		limit = { NOT = { has_variable = cmm_list_initialized_$mod_id$__$setting_id$ } }

		if = {
			limit = { var:cmm_list_add_counter_$mod_id$__$setting_id$ < 1 }
			set_variable = { name = cmm_list_add_counter_$mod_id$__$setting_id$ value = 1 }
		}

		set_variable = {
			name = cmm_setting_list_count_$mod_id$__$setting_id$
			value = var:cmm_list_add_counter_$mod_id$__$setting_id$
		}

		set_variable = {
			name = cmm_setting_list_last_index_$mod_id$__$setting_id$
			value = var:cmm_list_add_counter_$mod_id$__$setting_id$
		}
		change_variable = { name = cmm_setting_list_last_index_$mod_id$__$setting_id$ add = -1 }

		add_to_variable_map = { name = cmm_list_count key = flag:$mod_id$__$setting_id$ value = var:cmm_list_add_counter_$mod_id$__$setting_id$ }
		add_to_variable_map = { name = cmm_list_last_index key = flag:$mod_id$__$setting_id$ value = var:cmm_setting_list_last_index_$mod_id$__$setting_id$ }

		set_variable = { name = cmm_list_initialized_$mod_id$__$setting_id$ value = 1 }
	}

	cmm_suppress_list_setting_warnings = { mod_id = $mod_id$ setting_id = $setting_id$ }
}

# Register a settings list whose items are built from an existing country variable list.
# Each entry in `$list$` becomes one row; its scope is stored as the item value and is
# accessible as `scope:cmm_list_current_item_value` when iterating with cmm_for_each_list_item.
# Row labels use the default `<mod_id>__<setting_id>_item_<N>_name` localization key pattern.
# Register fields (cmm_register_list_bool_field, etc.) after this call.
# Idempotent: items are captured only on the first registration call.
# Scope: country
# Args:
# - mod_id (key): Owner mod id.
# - setting_id (key): Setting id unique within mod.
# - tab_id (key): Owner tab id within mod.
# - is_ordered (1/0): `1` shows row move controls, `0` hides them.
# - list (key): Name of the country variable list to iterate (max 20 entries used).
cmm_register_settings_list_from_list = {
	cmm_begin_settings_list = {
		mod_id = $mod_id$
		setting_id = $setting_id$
		tab_id = $tab_id$
		is_ordered = $is_ordered$
	}

	if = {
		limit = { NOT = { has_variable = cmm_list_initialized_$mod_id$__$setting_id$ } }
		every_in_list = {
			variable = $list$
			save_scope_as = cmm_list_staged_item_value
			root = {
				cmm_add_staged_settings_list_item = {
					mod_id = $mod_id$
					setting_id = $setting_id$
				}
			}
		}
	}

	cmm_finish_settings_list = { mod_id = $mod_id$ setting_id = $setting_id$ }
}
