# Add a building type to the list of types to hide in the global scope
# ROOT: none
# ARGS: building_type
cmf_hide_building = {
	add_to_global_variable_list = {
		name = cmf_global_hidden_buildings
		target = $building_type$
	}
}

# Remove a building type from the list of types to hide in the global scope
# ROOT: none
# ARGS: building_type
cmf_show_building = {
	remove_list_global_variable = {
		name = cmf_global_hidden_buildings
		target = $building_type$
	}
}
