# Register an action bar element globally and for all current countries.
# Scope: country
# Args: element (flag)
cmf_add_action_bar_element = {
    hidden_effect = {
        add_to_global_variable_list = {
            name = cmf_action_bar_elements
            target = flag:$element$
        }
        every_country = {
            add_to_variable_list = {
                name = cmf_action_bar_active_elements
                target = flag:$element$
            }
            add_to_variable_map = { name = cmf_ab_has_sgui key = flag:$element$ value = 0 }
        }
    }

    # Benign warning suppression
    if = {
        limit = {
            always = no
            exists = flag:$element$
            exists = global_var:cmf_action_bar_elements
            has_variable_map = cmf_ab_has_sgui
        }
    }
}

# Remove an action bar element globally and from all countries.
# Scope: country
# Args: element (flag)
cmf_remove_action_bar_element = {
    hidden_effect = {
        remove_list_global_variable = {
            name = cmf_action_bar_elements
            target = flag:$element$
        }
        every_country = {
            remove_list_variable = {
                name = cmf_action_bar_active_elements
                target = flag:$element$
            }
            remove_from_variable_map = { name = cmf_ab_has_sgui key = flag:$element$ }
        }
    }
}

# Register that an action bar element has a scripted GUI for is_shown/is_valid.
# Call after cmf_add_action_bar_element. The scripted GUI must be named the same as the element.
# Scope: country
# Args: element (flag)
cmf_register_action_bar_element_gui = {
    hidden_effect = {
        every_country = {
            remove_from_variable_map = { name = cmf_ab_has_sgui key = flag:$element$ }
            add_to_variable_map = { name = cmf_ab_has_sgui key = flag:$element$ value = 1 }
        }
    }
}
