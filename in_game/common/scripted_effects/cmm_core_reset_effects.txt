# Restore one key's value in the cmm value map from its registered default.
# Scope: country
# Args:
# - key_scope (key): Saved scope name holding the map key.
_cmm_restore_default_for_key = {
	# Macros do not resolve inside quoted variable_map lookups.
	scope:$key_scope$ = { save_scope_as = _cmm_reset_active_key }

	# List-field keys store their default in cmm_list_field_default, plain settings in cmm_default.
	if = {
		limit = { is_key_in_variable_map = { name = cmm_list_field_default target = scope:_cmm_reset_active_key } }
		set_local_variable = { name = cmm_reset_default value = "variable_map(cmm_list_field_default|scope:_cmm_reset_active_key)" }
	}
	else = {
		set_local_variable = { name = cmm_reset_default value = "variable_map(cmm_default|scope:_cmm_reset_active_key)" }
	}

	if = {
		limit = { "variable_map(cmm_is_global|scope:_cmm_reset_active_key)" >= 1 }
		remove_from_global_variable_map = { name = cmm key = scope:_cmm_reset_active_key }
		add_to_global_variable_map = { name = cmm key = scope:_cmm_reset_active_key value = local_var:cmm_reset_default }
	}
	else = {
		remove_from_variable_map = { name = cmm key = scope:_cmm_reset_active_key }
		add_to_variable_map = { name = cmm key = scope:_cmm_reset_active_key value = local_var:cmm_reset_default }
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = cmm_reset_default }
	}
}

# Fire cmf_on_callback for the setting key in scope:cmm_reset_key.
# Scope: country
_cmm_fire_reset_callback = {
	set_variable = { name = cmf_callback value = scope:cmm_reset_key }
	trigger_event_silently = {
		on_action = cmf_on_callback
	}
	remove_variable = cmf_callback
}

# Reset the setting key in scope:cmm_reset_key to its registered defaults and fire cmf_on_callback.
# Plain settings restore from cmm_default; list settings restore each owned
# item-field key from cmm_list_field_default. Settings marked cmm_no_reset and
# global settings of a non-host country are skipped.
# Scope: country
_cmm_reset_setting_to_default = {
	if = {
		limit = {
			is_key_in_variable_map = {
				name = cmm_default
				target = scope:cmm_reset_key
			}
			NOT = {
				is_key_in_variable_map = {
					name = cmm_no_reset
					target = scope:cmm_reset_key
				}
			}
			OR = {
				"variable_map(cmm_is_global|scope:cmm_reset_key)" < 1
				is_host = yes
			}
		}

		_cmm_restore_default_for_key = { key_scope = cmm_reset_key }
		_cmm_fire_reset_callback = yes
	}
	else_if = {
		limit = {
			"variable_map(cmm_type|scope:cmm_reset_key)" = 7
			NOT = {
				is_key_in_variable_map = {
					name = cmm_no_reset
					target = scope:cmm_reset_key
				}
			}
			OR = {
				"variable_map(cmm_is_global|scope:cmm_reset_key)" < 1
				is_host = yes
			}
		}

		clear_variable_list = _cmm_tmp_reset_field_keys
		every_key_in_variable_map = {
			variable = cmm_list_field_default

			save_scope_as = cmm_reset_field_key

			root = {
				if = {
					limit = {
						is_key_in_variable_map = {
							name = cmm_list_field_owner
							target = scope:cmm_reset_field_key
						}
						"variable_map(cmm_list_field_owner|scope:cmm_reset_field_key)" = scope:cmm_reset_key
					}
					add_to_variable_list = { name = _cmm_tmp_reset_field_keys target = scope:cmm_reset_field_key }
				}
			}
		}

		every_in_list = {
			variable = _cmm_tmp_reset_field_keys

			save_scope_as = cmm_reset_field_key

			root = {
				_cmm_restore_default_for_key = { key_scope = cmm_reset_field_key }
			}
		}

		clear_variable_list = _cmm_tmp_reset_field_keys

		_cmm_fire_reset_callback = yes
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = _cmm_tmp_reset_field_keys }
	}
}

# Reset all registered settings to their registered default values.
# Scope: country
_cmm_reset_to_defaults = {
	every_in_list = {
		variable = cmm_registered_setting_keys

		save_scope_as = cmm_reset_key

		root = {
			_cmm_reset_setting_to_default = yes
		}
	}
}

# Reset all registered settings owned by the mod in scope:mod to their registered default values.
# Scope: country
_cmm_reset_mod_to_defaults = {
	every_in_list = {
		variable = cmm_registered_setting_keys

		save_scope_as = cmm_reset_key

		root = {
			if = {
				limit = { "variable_map(cmm_setting_owner_mod|scope:cmm_reset_key)" = scope:mod }
				_cmm_reset_setting_to_default = yes
			}
		}
	}
}
