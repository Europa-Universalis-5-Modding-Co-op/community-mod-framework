#noinspection ALL
# True when the menu fired this change with no player country to apply it to.
# Scope: any
_cmm_is_staging_active = {
	trigger_if = {
		limit = { exists = scope:cmm_staging }
		scope:cmm_staging = yes
	}
	trigger_else = {
		always = no
	}
}

# True when scope:setting must be written to the staging map instead of a country.
# Scope: country
_cmm_is_staging_write = {
	_cmm_is_staging_active = yes
	"variable_map(cmm_is_global|scope:setting)" < 1
}
