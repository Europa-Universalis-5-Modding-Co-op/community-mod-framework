#noinspection ALL
# True when this country is controlled by the host player, or in singleplayer.
# Scope: country
is_host = {
	trigger_if = {
		limit = { has_multiple_players = no }
		is_human = yes
	}
	trigger_else = {
		trigger_if = {
			limit = { has_global_variable = cmf_core_host_country }
			this ?= global_var:cmf_core_host_country
		}
		trigger_else = {
			always = no
		}
	}
}

# True when this country can use unrestricted tools: the unrestricted-tools setting is on and this country is the host (or in singleplayer).
# Gate for restricted features not meant for normal play.
# Scope: country
cmf_can_use_unrestricted_tools = {
	trigger_if = {
		limit = {
			has_global_variable_map = cmm
			is_key_in_global_variable_map = { name = cmm target = flag:cmf_core__enable_unrestricted_tools }
		}
		"global_variable_map(cmm|flag:cmf_core__enable_unrestricted_tools)" >= 1
		is_host = yes
	}
	trigger_else = {
		always = no
	}
}

# True when a mod with the given id is active in the current game.
# Scope: any
# Args: mod_id (key)
cmf_is_mod_active = {
	hidden_trigger = {
		is_target_in_global_variable_list = {
			name = cmf_active_mod_ids
			target = flag:$mod_id$
		}
	}
}

# True when a mod with the given id was active when this game was started as a new game.
# Older saves predating this feature have no record and read false.
# Scope: any
# Args: mod_id (key)
cmf_was_mod_active_at_game_start = {
	hidden_trigger = {
		is_target_in_global_variable_list = {
			name = cmf_game_start_mod_ids
			target = flag:$mod_id$
		}
	}
}

# True when the game-start mod set was recorded (game created after this feature shipped).
# Scope: any
cmf_game_start_mods_recorded = {
	hidden_trigger = {
		has_global_variable = cmf_game_start_recorded
	}
}
