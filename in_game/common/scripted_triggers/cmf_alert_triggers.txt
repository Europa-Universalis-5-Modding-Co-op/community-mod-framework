# Check if an alert is active (shown by the mod; the player may have hidden or muted it).
# Scope: country
# Args: alert (key) - the alert root localization key
cmf_is_alert_active = {
    hidden_trigger = {
        is_target_in_variable_list = {
            name = cmf_active_alerts
            target = flag:$alert$
        }
    }
}
