# Run built-in CMM registration and the shared registration hook.
# Scope: country
CMM_RegisterCoreMod = {
	effect = {
		clear_variable_list = cmm_registered_mod_ids
		clear_variable_list = cmm_registered_tab_keys
		clear_variable_list = cmm_registered_group_keys
		clear_variable_list = cmm_registered_setting_keys
		clear_variable_list = cmm_conditional_setting_keys
		_cmm_clear_registration_pass_state = yes
		clear_global_variable_list = cmf_active_mod_ids
		_cmm_register_menu_settings = yes
		trigger_event_silently = {
			on_action = cmf_on_mod_registration
		}
		_cmf_prune_stale_action_bar_elements = yes
		_cmm_register_menu_settings_late = yes
	}
}

# List _on_changed for the action-bar visible-buttons list.
# Mirrors the toggled row to cmf_action_bar_active_elements before applying
# the change to the bool field's stored value.
# Scope: country
cmf_core__action_bar_buttons_on_changed = {
	scope = country

	effect = {
		if = {
			limit = {
				has_variable = cmm_list_position_to_apply
				has_variable = cmm_list_field_slot_to_apply
				has_variable = cmm_list_field_value_to_apply
				var:cmm_list_field_slot_to_apply = 1
			}
			set_variable = { name = _cmm_action_bar_change_pos value = var:cmm_list_position_to_apply }
			set_variable = { name = _cmm_action_bar_change_val value = var:cmm_list_field_value_to_apply }

			cmm_for_each_list_item = {
				setting = cmf_core__action_bar_buttons
				effect = _cmm_handle_action_bar_button_change
			}

			remove_variable = _cmm_action_bar_change_pos
			remove_variable = _cmm_action_bar_change_val
		}

		cmm_apply_list_change = {
			setting = cmf_core__action_bar_buttons
		}

		# Benign warning suppression
		if = {
			limit = { always = no }
			cmf_suppress = { v = _cmm_action_bar_change_pos }
			cmf_suppress = { v = _cmm_action_bar_change_val }
		}
	}
}
