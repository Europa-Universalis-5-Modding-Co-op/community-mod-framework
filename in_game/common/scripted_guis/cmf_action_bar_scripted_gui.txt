# GetScriptedGui dispatch stand-in while CMFActionBarSynced is false; behaves
# like an element with no scripted GUI.
# Scope: country
cmf_action_bar_fallback = {
	scope = country

	is_shown = {
		always = yes
	}

	is_valid = {
		always = yes
	}
}

# Auto-apply: handle action bar button left-click. Fires cmf_on_callback.
# Scope: country
# Args: scope:cmf_element (flag) - the action bar element root key.
cmf_auto_action_bar_clicked = {
	scope = country
	saved_scopes = { cmf_element }

	effect = {
		set_variable = { name = cmf_callback value = scope:cmf_element }
		trigger_event_silently = {
			on_action = cmf_on_callback
		}
		remove_variable = cmf_callback

		# Benign warning suppression
		if = {
			limit = { always = no }
			cmf_suppress = { v = cmf_callback }
		}
	}
}
