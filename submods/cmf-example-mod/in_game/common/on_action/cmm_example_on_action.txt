# Hook this mod into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		cmm_example_on_register_mod
	}
}

cmm_example_on_register_mod = {
	effect = {
		# CMM settings
		cmm_example_register_mod = yes

		# Action bar elements (idempotent — safe from per-country hook)
		cmf_add_action_bar_element = { element = cmf_action_bar_element_example }
		cmf_add_action_bar_element = { element = cmf_action_bar_element_example_second }

		# Alerts
		cmf_add_alert = { alert = cmf_alert_example }
		cmf_add_alert = { alert = cmf_alert_example_blue }
		cmf_add_alert = { alert = cmf_alert_example_black }
		cmf_add_alert = { alert = cmf_alert_example_red_war }
		cmf_add_alert = { alert = cmf_alert_example_red }
		cmf_add_alert = { alert = cmf_alert_example_orange }
		cmf_add_alert = { alert = cmf_alert_example_yellow }
		cmf_add_alert = { alert = cmf_alert_example_green }
		cmf_add_alert = { alert = cmf_alert_example_purple }
	}
}

# Unified callback hook — handles setting changes, alert clicks, and action bar clicks.
cmf_on_callback = {
	on_actions = {
		cmm_example_on_callback
	}
}

cmm_example_on_callback = {
	effect = {
		# Toggle the example alert when the example toggle is changed.
		if = {
			limit = { var:cmf_callback = flag:cmm_example__example_toggle }
			set_local_variable = { name = cmm_tmp_flag value = flag:cmm_example__example_toggle }
			if = {
				limit = { "variable_map(cmm|local_var:cmm_tmp_flag)" >= 1 }
				cmf_add_alert = { alert = cmf_alert_example }
			}
			else = {
				cmf_remove_alert = { alert = cmf_alert_example }
			}
		}

		# Benign warning suppression
		if = {
			limit = {
				always = no
				exists = local_var:cmm_tmp_flag
			}
		}
	}
}
