# Register action bar elements and alerts.
# Scope: country
cmf_feature_test_register_cmf = {
	# Action bar elements (idempotent)
	cmf_add_action_bar_element = { element = cmf_action_bar_ft_primary }
	cmf_add_action_bar_element = { element = cmf_action_bar_ft_secondary }

	# Register scripted GUI for primary button (enables is_shown/is_valid)
	cmf_register_scripted_gui = { element = cmf_action_bar_ft_primary }

	# Alerts
	cmm_register_bool_setting = {
		mod_id = cmf_feature_test
		setting_id = alerts_enabled
		tab_id = settings
		group_id = alerts
		default_value = 0
	}
	cmf_feature_test_sync_alerts = yes
}

# Sync all test alerts to the Test Alerts setting.
# Scope: country
cmf_feature_test_sync_alerts = {
	if = {
		limit = { "variable_map(cmm|flag:cmf_feature_test__alerts_enabled)" >= 1 }
		cmf_show_alert = { alert = cmf_alert_ft_orange }
		cmf_show_alert = { alert = cmf_alert_ft_blue }
		cmf_show_alert = { alert = cmf_alert_ft_black }
		cmf_show_alert = { alert = cmf_alert_ft_red_war }
		cmf_show_alert = { alert = cmf_alert_ft_red }
		cmf_show_alert = { alert = cmf_alert_ft_yellow }
		cmf_show_alert = { alert = cmf_alert_ft_green }
		cmf_show_alert = { alert = cmf_alert_ft_purple }
	}
	else = {
		cmf_remove_alert = { alert = cmf_alert_ft_orange }
		cmf_remove_alert = { alert = cmf_alert_ft_blue }
		cmf_remove_alert = { alert = cmf_alert_ft_black }
		cmf_remove_alert = { alert = cmf_alert_ft_red_war }
		cmf_remove_alert = { alert = cmf_alert_ft_red }
		cmf_remove_alert = { alert = cmf_alert_ft_yellow }
		cmf_remove_alert = { alert = cmf_alert_ft_green }
		cmf_remove_alert = { alert = cmf_alert_ft_purple }
	}
}

# Unified callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
cmf_feature_test_handle_callback = {
	switch = {
		trigger = var:cmf_callback

		# -- Setting callbacks ---------------------------------------------

		flag:cmf_feature_test__alerts_enabled = {
			cmf_feature_test_sync_alerts = yes
		}

		flag:cmf_feature_test__ordered_list = {
			# Iterate list items to verify cmm_for_each_list_item works
			cmm_for_each_list_item = {
				setting = cmf_feature_test__ordered_list
				effect = cmf_feature_test_process_list_item
			}
		}

		# -- Alert callbacks -----------------------------------------------
		# Re-add alerts after they are clicked so they reappear for repeated testing.

		flag:cmf_alert_ft_orange = {
			cmf_show_alert = { alert = cmf_alert_ft_orange }
		}
		flag:cmf_alert_ft_blue = {
			cmf_show_alert = { alert = cmf_alert_ft_blue }
		}
		flag:cmf_alert_ft_black = {
			cmf_show_alert = { alert = cmf_alert_ft_black }
		}
		flag:cmf_alert_ft_red_war = {
			cmf_show_alert = { alert = cmf_alert_ft_red_war }
		}
		flag:cmf_alert_ft_red = {
			cmf_show_alert = { alert = cmf_alert_ft_red }
		}
		flag:cmf_alert_ft_yellow = {
			cmf_show_alert = { alert = cmf_alert_ft_yellow }
		}
		flag:cmf_alert_ft_green = {
			cmf_show_alert = { alert = cmf_alert_ft_green }
		}
		flag:cmf_alert_ft_purple = {
			cmf_show_alert = { alert = cmf_alert_ft_purple }
		}

		# -- Action bar callbacks ------------------------------------------

		flag:cmf_action_bar_ft_primary = {
			# Toggle the blue alert to demonstrate action bar interaction
			if = {
				limit = { cmf_is_alert_active = { alert = cmf_alert_ft_blue } }
				cmf_remove_alert = { alert = cmf_alert_ft_blue }
			}
			else_if = {
				limit = { "variable_map(cmm|flag:cmf_feature_test__alerts_enabled)" >= 1 }
				cmf_show_alert = { alert = cmf_alert_ft_blue }
			}
		}

		flag:cmf_action_bar_ft_secondary = {
			# Toggle the green alert to demonstrate action bar interaction
			if = {
				limit = { cmf_is_alert_active = { alert = cmf_alert_ft_green } }
				cmf_remove_alert = { alert = cmf_alert_ft_green }
			}
			else_if = {
				limit = { "variable_map(cmm|flag:cmf_feature_test__alerts_enabled)" >= 1 }
				cmf_show_alert = { alert = cmf_alert_ft_green }
			}
		}
	}
}
