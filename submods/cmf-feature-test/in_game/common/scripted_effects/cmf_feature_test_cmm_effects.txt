# Helper effect called by cmm_for_each_list_item for the ordered list.
# $i$ = item number (1-based, in display order), $setting$ = composite key.
# scope:cmm_list_current_item_value = attached game scope (root country).
# Scope: country
cmf_feature_test_process_list_item = {
	# Read field values for this item to verify the iteration API works.
	# Use local variable indirection to avoid variable_map flag: parse issues.
	set_local_variable = { name = cft_fk_1 value = flag:$setting$_i$i$_f1 }
	set_local_variable = { name = cft_fk_2 value = flag:$setting$_i$i$_f2 }
	set_local_variable = { name = cft_fk_3 value = flag:$setting$_i$i$_f3 }
	set_local_variable = {
		name = cft_test_field_1
		value = "variable_map(cmm|local_var:cft_fk_1)"
	}
	set_local_variable = {
		name = cft_test_field_2
		value = "variable_map(cmm|local_var:cft_fk_2)"
	}
	set_local_variable = {
		name = cft_test_field_3
		value = "variable_map(cmm|local_var:cft_fk_3)"
	}

	# Benign warning suppression
	if = {
		limit = {
			always = no
			exists = local_var:cft_fk_1
			exists = local_var:cft_fk_2
			exists = local_var:cft_fk_3
			exists = local_var:cft_test_field_1
			exists = local_var:cft_test_field_2
			exists = local_var:cft_test_field_3
		}
	}
}

# Suppress "set but never used" for from-list variable list target scopes.
# These user-chosen flags can't be suppressed by the core.
cmf_feature_test_suppress_from_list_flags = {
	if = {
		limit = {
			always = no
			exists = flag:cft_item_alpha
			exists = flag:cft_item_beta
			exists = flag:cft_item_gamma
			exists = flag:cft_global_item_alpha
			exists = flag:cft_global_item_beta
			exists = flag:cft_global_item_gamma
		}
	}
}
