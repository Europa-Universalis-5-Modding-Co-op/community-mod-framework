# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
cmf_feature_test_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Settings (settings) Tab
		## Toggles (toggles) Group
		flag:cmf_feature_test__toggle = {
			cmm_sync_bool_alias = {
				setting = cmf_feature_test__toggle
				alias = cft_toggle_alias
			}
		}
	}
}
