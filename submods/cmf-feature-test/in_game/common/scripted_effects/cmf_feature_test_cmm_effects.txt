	cmm_sync_dropdown_option_alias = { setting = cmf_feature_test__dropdown index = 5 alias = cft_dd_opt_5 }

	cmf_feature_test_suppress_from_list_flags = yes
}

# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
cmf_feature_test_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Settings (settings) Tab
		## Toggles (toggles) Group
		flag:cmf_feature_test__toggle = {
			cmm_sync_bool_alias = {
				setting = cmf_feature_test__toggle
				alias = cft_toggle_alias
			}
		}
		flag:cmf_feature_test__log_test = {
			cmf_log = { action = cft_log_test_action }
		}
		flag:cmf_feature_test__log_test_args = {
			cmf_log_with_args = { action = cft_log_transferred_to arg1 = cft_log_test_location arg2 = cft_log_test_country }
		}
		flag:cmf_feature_test__log_test_scope_args = {
			# Save two countries as scope args, then log with scope variant
			random_country = {
				limit = { NOT = { this = root } }
				save_scope_as = cmf_log_arg1
			}
			random_country = {
				limit = { NOT = { this = root } NOT = { this = scope:cmf_log_arg1 } }
				save_scope_as = cmf_log_arg2
			}
			cmf_log_with_scope_args = { action = cft_log_transferred_to }
		}
	}
}

# Helper effect called by cmm_for_each_list_item for the ordered list.
# Receives $i$ (item number) and $setting$ (composite key) as macro arguments.
# scope:cmm_list_current_item_value = attached game scope (root country).
# Scope: country
cmf_feature_test_process_list_item = {
	if = {
		limit = { exists = scope:cmm_list_current_item_value }
		save_scope_as = cft_test_item_value
	}

	# Benign warning suppression - references $i$ and $setting$ so the engine
	# recognizes this as a parameterized effect (required by cmm_for_each_list_item).
	if = {
		limit = { always = no }
		cmf_suppress = { v = cft_item_index }
		cmf_suppress = { v = cft_setting_key }
		set_local_variable = { name = cft_item_index value = $i$ }
		set_local_variable = { name = cft_setting_key value = flag:$setting$ }
		save_scope_as = cft_test_item_value
	}
}

# Suppress "set but never used" for from-list variable list target flags.
# These user-chosen flags can't be suppressed by the core.
cmf_feature_test_suppress_from_list_flags = {
	if = {
		limit = { always = no }
		cmf_suppress = { v = cft_item_alpha }
		cmf_suppress = { v = cft_item_beta }
		cmf_suppress = { v = cft_item_gamma }
		cmf_suppress = { v = cft_global_item_alpha }
		cmf_suppress = { v = cft_global_item_beta }
		cmf_suppress = { v = cft_global_item_gamma }
		cmf_suppress = { v = cft_log_test_action }
		cmf_suppress = { v = cft_log_transferred_to }
		cmf_suppress = { v = cft_log_test_location }
		cmf_suppress = { v = cft_log_test_country }
	}
}
