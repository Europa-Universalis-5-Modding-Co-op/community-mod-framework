# Hook into CMF shared registration on_action.
cmf_on_mod_registration = {
	on_actions = {
		cmf_feature_test_on_register_mod
	}
}

cmf_feature_test_on_register_mod = {
	effect = {
		# CMM settings (all tabs, all types, all list variants, aliases)
		cmf_feature_test_register_cmm = yes

		# CMF features (action bar elements and alerts)
		cmf_feature_test_register_cmf = yes
	}
}

# Unified callback hook — handles setting changes, alert clicks, and action bar clicks.
cmf_on_callback = {
	on_actions = {
		cmf_feature_test_on_callback
	}
}

cmf_feature_test_on_callback = {
	effect = {
		cmf_feature_test_handle_callback = yes
	}
}
