# Post-lobby hook tests (no scope).
# on_game_start_after_lobby: changes France's pops to Albanian (new game only).
# on_game_load: changes England's pops to Bacan (every save load).
# on_game_load_after_lobby: changes Castile's pops to Cabecar (save load, past lobby only).
on_game_start_after_lobby = {
	on_actions = {
		cmf_feature_test_on_game_start_after_lobby
	}
}

cmf_feature_test_on_game_start_after_lobby = {
	effect = {
		c:FRA = {
			every_owned_location = {
				every_pop = {
					change_pop_culture = culture:albanian
				}
				force_refresh_culture_and_religion = this
			}
		}
	}
}

on_game_load = {
	on_actions = {
		cmf_feature_test_on_game_load
	}
}

cmf_feature_test_on_game_load = {
	effect = {
		c:ENG = {
			every_owned_location = {
				every_pop = {
					change_pop_culture = culture:bacan_culture
				}
				force_refresh_culture_and_religion = this
			}
		}
	}
}

on_game_load_after_lobby = {
	on_actions = {
		cmf_feature_test_on_game_load_after_lobby
	}
}

cmf_feature_test_on_game_load_after_lobby = {
	effect = {
		c:CAS = {
			every_owned_location = {
				every_pop = {
					change_pop_culture = culture:cabecar_culture
				}
				force_refresh_culture_and_religion = this
			}
		}
	}
}

# Post-lobby hook tests (human country scope).
# on_game_start_after_lobby_human_country: changes human pops to Danish (new game only).
# on_game_load_human_country: changes human pops to English (every save load).
# on_game_load_after_lobby_human_country: changes human pops to French (save load, past lobby only).
on_game_start_after_lobby_human_country = {
	on_actions = {
		cmf_feature_test_on_game_start_after_lobby_human_country
	}
}

cmf_feature_test_on_game_start_after_lobby_human_country = {
	effect = {
		every_owned_location = {
			every_pop = {
				change_pop_culture = culture:danish
			}
			force_refresh_culture_and_religion = this
		}
	}
}

on_game_load_human_country = {
	on_actions = {
		cmf_feature_test_on_game_load_human_country
	}
}

cmf_feature_test_on_game_load_human_country = {
	effect = {
		every_owned_location = {
			every_pop = {
				change_pop_culture = culture:english
			}
			force_refresh_culture_and_religion = this
		}
	}
}

on_game_load_after_lobby_human_country = {
	on_actions = {
		cmf_feature_test_on_game_load_after_lobby_human_country
		cmf_feature_test_log_game_start_mods
	}
}

cmf_feature_test_on_game_load_after_lobby_human_country = {
	effect = {
		every_owned_location = {
			every_pop = {
				change_pop_culture = culture:french
			}
			force_refresh_culture_and_religion = this
		}
	}
}

# Logs whether Feature Test was in the game-start snapshot, to confirm it survives a save/reload.
cmf_feature_test_log_game_start_mods = {
	effect = {
		if = {
			limit = { NOT = { cmf_game_start_mods_recorded = yes } }
			cmf_log = { action = cft_log_game_start_unrecorded }
		}
		else_if = {
			limit = { cmf_was_mod_active_at_game_start = { mod_id = cmf_feature_test } }
			cmf_log = { action = cft_log_game_start_active }
		}
		else = {
			cmf_log = { action = cft_log_game_start_inactive }
		}
	}
}
