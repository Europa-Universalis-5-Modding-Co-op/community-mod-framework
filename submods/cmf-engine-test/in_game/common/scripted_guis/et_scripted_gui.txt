# Engine validation scripted GUI.
# Receives scope:et_setting via AddScope from the GUI button.
# Tests three assumptions about variable maps + AddScope interaction.
et_run = {
	scope = country

	saved_scopes = {
		et_setting
	}

	effect = {
		# Reset all results before running
		set_variable = { name = et_1_passed value = 0 }
		set_variable = { name = et_2_passed value = 0 }
		set_variable = { name = et_3_passed value = 0 }

		# Result: PASS
		# -- Test 1 ----------------------------------------------
		# Can scope:et_setting (a flag passed via AddScope) be used
		# as a variable map key?
		# If this line crashes, all three tests stay at 0 (fail).
		add_to_variable_map = {
			name = et_map
			key = scope:et_setting
			value = 42
		}
		set_variable = { name = et_1_passed value = 1 }

		# Result: PASS
		# -- Test 2 ----------------------------------------------
		# Can "variable_map(name|scope:et_setting)" resolve the
		# stored value back through the same scope reference?
		if = {
			limit = {
				"variable_map(et_map|scope:et_setting)" >= 42
			}
			set_variable = { name = et_2_passed value = 1 }
		}

		# Result: PASS
		# -- Test 3 ----------------------------------------------
		# Can the map value be stored in a local variable and
		# used for arithmetic?
		set_local_variable = {
			name = temp
			value = "variable_map(et_map|scope:et_setting)"
		}
		change_local_variable = {
			name = temp
			add = 1
		}
		if = {
			limit = { local_var:temp >= 43 }
			set_variable = { name = et_3_passed value = 1 }
		}

		# -- Seeds for cross-context tests (run by global suite) --
		# Write with hardcoded flag: key. Global suite will try to
		# update these entries using scope:et_setting.
		add_to_variable_map = {
			name = et_cross
			key = flag:et_key_a
			value = 55
		}
		add_to_global_variable_map = {
			name = et_gmap_update
			key = flag:et_key_a
			value = 10
		}
		add_to_variable_map = {
			name = et_cmap_update
			key = flag:et_key_a
			value = 10
		}

		# Mark that the test suite has been executed
		set_variable = { name = et_has_run value = 1 }
	}
}

				global_var:et_nb_to_done = 1
			}
			global_var:et_nb_from = {
				save_temporary_scope_as = et_nb_a
				every_neighbor_location = {
					save_temporary_scope_as = et_nb_n
					if = {
						limit = { scope:et_nb_a = { has_latest_road_to = scope:et_nb_n } }
						change_global_variable = { name = et_nb_rd_a2 add = 1 }
					}
				}
				if = {
					limit = { has_road_constructions = no }
					set_global_variable = { name = et_175_ok value = 1 }
				}
			}
			global_var:et_nb_to = {
				save_temporary_scope_as = et_nb_b
				every_neighbor_location = {
					save_temporary_scope_as = et_nb_n
					if = {
						limit = { scope:et_nb_b = { has_latest_road_to = scope:et_nb_n } }
						change_global_variable = { name = et_nb_rd_b2 add = 1 }
					}
				}
			}
			if = {
				limit = { global_var:et_nb_rd_a2 > global_var:et_nb_rd_a }
				set_variable = { name = et_173_passed value = 1 }
			}
			if = {
				limit = { global_var:et_nb_rd_b2 > global_var:et_nb_rd_b }
				set_variable = { name = et_174_passed value = 1 }
			}
			if = {
				limit = { global_var:et_175_ok = 1 }
				set_variable = { name = et_175_passed value = 1 }
			}
		}

		# 159: the capital's plain instant = yes levels.
		capital = {
			if = {
				limit = {
					any_buildings_in_location = {
						building_type = building_type:et_probe_multi
						building_levels_under_construction = 0
					}
				}
				set_global_variable = { name = et_159_ok value = 1 }
			}
		}

		# 160: the cost_multiplier = 0 plus instant = yes build at the second location, which
		# nothing else touches. This is the lucky-nation governor relocation's exact call.
		if = {
			limit = { global_var:et_uc_done = 1 }
			global_var:et_uc_loc = {
				if = {
					limit = {
						any_buildings_in_location = {
							building_type = building_type:et_probe_multi
							building_level >= 1
							building_levels_under_construction = 0
						}
					}
					set_global_variable = { name = et_160_ok value = 1 }
				}
			}
		}

		if = {
			limit = { global_var:et_159_ok = 1 }
			set_variable = { name = et_159_passed value = 1 }
		}
		if = {
			limit = { global_var:et_160_ok = 1 }
			set_variable = { name = et_160_passed value = 1 }
		}

		# 164-168: the max-level reads again after game time has passed. Every immediate read has
		# come back 0 - probe and vanilla types, with and without the owner param - and a value
		# that only refreshes on a month tick would look exactly like that.
		set_global_variable = { name = et_ml_c value = 0 }
		set_global_variable = { name = et_165_ok value = 0 }
		set_global_variable = { name = et_166_ok value = 0 }
		set_global_variable = { name = et_167_ok value = 0 }
		capital = {
			if = {
				limit = { building_type_max_level = { building_type = building_type:et_probe_multi value = 0 } }
				set_global_variable = { name = et_ml_c value = 1 }
			}
			else_if = {
				limit = { building_type_max_level = { building_type = building_type:et_probe_multi value = 3 } }
				set_global_variable = { name = et_ml_c value = 3 }
			}
			else = {
				set_global_variable = { name = et_ml_c value = 4 }
			}
			if = {
				limit = { building_type_max_level = { building_type = building_type:et_probe_multi owner = owner value = 3 } }
				set_global_variable = { name = et_165_ok value = 1 }
			}
			if = {
				limit = { building_type_max_level = { building_type = building_type:mason value > 0 } }
				set_global_variable = { name = et_166_ok value = 1 }
			}
			if = {
				limit = { building_type_max_level = { building_type = building_type:royal_court value = 1 } }
				set_global_variable = { name = et_167_ok value = 1 }
			}
		}
		if = {
			limit = { global_var:et_ml_c = 3 }
			set_variable = { name = et_164_passed value = 1 }
		}
		if = {
			limit = { global_var:et_ml_c = 1 }
			set_variable = { name = et_168_passed value = 1 }
		}
		if = {
			limit = { global_var:et_165_ok = 1 }
			set_variable = { name = et_165_passed value = 1 }
		}
		if = {
			limit = { global_var:et_166_ok = 1 }
			set_variable = { name = et_166_passed value = 1 }
		}
		if = {
			limit = { global_var:et_167_ok = 1 }
			set_variable = { name = et_167_passed value = 1 }
		}

		# 190: whether the strait order 187-189 placed ever laid a road. A gravel road is
		# build_time_per_unit_distance 15 per unit of distance, so this needs months rather than
		# the days the rest of this pass wants, and a FAIL only means "not yet" until enough of
		# them have passed. Press again later; it reads live state and can be pressed repeatedly.
		set_variable = { name = et_190_passed value = 0 }
		if = {
			limit = { location:coruna = { has_road_to = location:ferrol } }
			set_variable = { name = et_190_passed value = 1 }
		}

		# 194-198 again, in case the engine keeps a per-country market list it rebuilds on a tick
		# rather than answering from the subject relationship live. The vassal is not re-made here,
		# so this reads the same setup the run button left.
		set_variable = { name = et_194_passed value = 0 }
		set_variable = { name = et_195_passed value = 0 }
		set_variable = { name = et_196_passed value = 0 }
		set_variable = { name = et_197_passed value = 0 }
		set_variable = { name = et_198_passed value = 0 }
		et_mk_measure = yes

		set_variable = { name = et_recheck_has_run value = 1 }
	}
