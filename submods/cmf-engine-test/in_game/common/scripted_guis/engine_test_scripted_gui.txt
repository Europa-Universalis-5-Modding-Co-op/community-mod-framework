# Engine validation scripted GUI.
# Receives scope:setting via AddScope from the GUI button.
# Tests three assumptions about variable maps + AddScope interaction.
engine_test_run = {
	scope = country

	saved_scopes = {
		setting
	}

	effect = {
		# Reset all results before running
		set_variable = { name = engine_test_1_passed value = 0 }
		set_variable = { name = engine_test_2_passed value = 0 }
		set_variable = { name = engine_test_3_passed value = 0 }

		# Result: PASS
		# -- Test 1 ----------------------------------------------
		# Can scope:setting (a flag passed via AddScope) be used
		# as a variable map key?
		# If this line crashes, all three tests stay at 0 (fail).
		add_to_variable_map = {
			name = engine_test_map
			key = scope:setting
			value = 42
		}
		set_variable = { name = engine_test_1_passed value = 1 }

		# Result: PASS
		# -- Test 2 ----------------------------------------------
		# Can "variable_map(name|scope:setting)" resolve the
		# stored value back through the same scope reference?
		if = {
			limit = {
				"variable_map(engine_test_map|scope:setting)" >= 42
			}
			set_variable = { name = engine_test_2_passed value = 1 }
		}

		# Result: PASS
		# -- Test 3 ----------------------------------------------
		# Can the map value be stored in a local variable and
		# used for arithmetic?
		set_local_variable = {
			name = temp
			value = "variable_map(engine_test_map|scope:setting)"
		}
		change_local_variable = {
			name = temp
			add = 1
		}
		if = {
			limit = { local_var:temp >= 43 }
			set_variable = { name = engine_test_3_passed value = 1 }
		}

		# -- Seeds for cross-context tests (run by global suite) --
		# Write with hardcoded flag: key. Global suite will try to
		# update these entries using scope:setting.
		add_to_variable_map = {
			name = engine_test_cross
			key = flag:engine_test_key_a
			value = 55
		}
		add_to_global_variable_map = {
			name = engine_test_gmap_update
			key = flag:engine_test_key_a
			value = 10
		}
		add_to_variable_map = {
			name = engine_test_cmap_update
			key = flag:engine_test_key_a
			value = 10
		}

		# Mark that the test suite has been executed
		set_variable = { name = engine_test_has_run value = 1 }
	}
}
