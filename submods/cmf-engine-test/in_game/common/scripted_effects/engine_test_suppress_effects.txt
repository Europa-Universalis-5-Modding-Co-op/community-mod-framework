# Dead branch suppression for engine test warnings.
# The script checker cannot see variable/flag usage in GUI expressions
# or inside quoted scope link strings. This unused effect references
# all flagged names so the checker considers them used/set.

	if = {
		limit = {
			always = no
			# Suppress "set but never used" for country variables

			var:engine_test_18_passed >= 0
			var:engine_test_19_passed >= 0
			var:engine_test_20_passed >= 0

			# Suppress "set but never used" for local variables
			local_var:engine_test_lv >= 0
			local_var:engine_test_map_name >= 0
		}
		# Suppress "used but never set" for prefixed variable references
		# (checker can't match quoted scope link usage to set_variable)
		set_variable = { name = engine_test_map_name_v value = 0 }
		set_local_variable = { name = engine_test_map_name value = 0 }
		# Suppress "set but never used" for flags (used as variable map keys)
		set_local_variable = { name = _s value = flag:engine_test_cross }
		set_local_variable = { name = _s value = flag:engine_test_hardcoded }
		set_local_variable = { name = _s value = flag:engine_test_iter_a }
		set_local_variable = { name = _s value = flag:engine_test_iter_b }
		set_local_variable = { name = _s value = flag:engine_test_iter_c }
		set_local_variable = { name = _s value = flag:engine_test_preseeded }
	}
}
