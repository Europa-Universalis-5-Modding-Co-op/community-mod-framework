# Scripted effects for the hidden-window engine tests (tests 69-71 probes)
# plus warning suppression for the new test variables.

# Sub-effect called from the OUTERMOST every_country loop in et_hw_finish.
# Scores test 69: a scope saved in the outermost loop is visible here.
et_hw_probe_outer = {
	if = {
		limit = { exists = scope:et_g69_outer }
		root = {
			set_variable = { name = et_69_passed value = 1 }
		}
	}
}

# Sub-effect called from the NESTED every_country loop in et_hw_finish.
# Test 70 (PASS 2026-07-02): the nested loop's saved scope WAS set here in
# this minimal shape; the known CMM failure came from an every_in_list +
# macro-param shape (see RESULTS.md). Test 71 (PASS): the local_var captured
# in the same loop body also survives into this sub-effect.
et_hw_probe_nested = {
	if = {
		limit = { exists = scope:et_g70_nested }
		root = {
			set_variable = { name = et_70_passed value = 1 }
		}
	}
	if = {
		limit = { exists = local_var:et_g71_lv }
		root = {
			set_variable = { name = et_71_passed value = 1 }
		}
	}
}
