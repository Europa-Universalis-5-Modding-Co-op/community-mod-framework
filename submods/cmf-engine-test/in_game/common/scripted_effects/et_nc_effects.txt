# on_action probes (tests 104-105). The script checker cannot see reads that
# happen in GUI expressions, so every name written here is referenced below.

			always = no

			exists = var:et_nc_fresh
			exists = var:et_nc_early_exec
			exists = var:et_nc_early_console
			exists = var:et_nc_early_armed

			exists = var:et_nc_tc_console
			exists = var:et_nc_to_exec
			exists = var:et_nc_to_console
			exists = global_var:et_gs_fires
			exists = global_var:et_wp_fires
		}
	}
}
