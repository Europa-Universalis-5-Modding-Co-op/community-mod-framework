on_game_start = {
	on_actions = {
		et_on_game_start
		et_on_game_start_load_probe
	}
}

et_on_game_start = {
	effect = {
		# Pre-seed global variable maps for read-only tests (8, 9).
		# These verify that scripted GUIs can read global maps even if
		# they can't write to them.
		# Note: on_game_start fires with empty scope, so set_variable
		# (country-scoped) cannot be used here. The scripted GUIs
		# reset their own result variables before running.
		add_to_global_variable_map = {
			name = et_gmap8
			key = flag:et_preseeded
			value = 10
		}
		add_to_global_variable_map = {
			name = et_gmap9
			key = flag:et_key_a
			value = 77
		}
	}
}

# Probe: does vanilla on_game_start fire on save load, or only on a new game? (see RESULTS.md)
et_on_game_start_load_probe = {
	effect = {
		if = {
			limit = { country_exists = c:FRA }
			c:FRA = {
				change_culture = culture:greek_culture
			}
		}
	}
}
